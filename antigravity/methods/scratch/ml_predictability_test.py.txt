import pandas as pd
import numpy as np
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score

df_ml = pd.read_csv(r'project_reports\colour_ml_ready_v17.csv')

# Feature matrix: 5 lags of numbers, colours, and sizes
# Encode features
features = []
for k in range(1, 6):
    df_ml[f'prev_{k}_num'] = df_ml[f'prev_{k}_number'].astype(int)
    df_ml[f'prev_{k}_is_big'] = (df_ml[f'prev_{k}_size'] == 'Big').astype(int)
    df_ml[f'prev_{k}_is_red'] = df_ml[f'prev_{k}_colour'].str.contains('Red').astype(int)
    df_ml[f'prev_{k}_is_green'] = df_ml[f'prev_{k}_colour'].str.contains('Green').astype(int)
    features.extend([f'prev_{k}_num', f'prev_{k}_is_big', f'prev_{k}_is_red', f'prev_{k}_is_green'])

X = df_ml[features].values
y_size = (df_ml['target_size'] == 'Big').astype(int).values
y_num = df_ml['target_number'].astype(int).values
y_col = df_ml['target_colour'].values

# Time-series chronological split: First 80% train, last 20% test
split_idx = int(len(X) * 0.8)
X_train, X_test = X[:split_idx], X[split_idx:]
y_size_train, y_size_test = y_size[:split_idx], y_size[split_idx:]
y_num_train, y_num_test = y_num[:split_idx], y_num[split_idx:]
y_col_train, y_col_test = y_col[:split_idx], y_col[split_idx:]

print(f"Train samples: {len(X_train)}, Test samples: {len(X_test)}")

# 1. Size Prediction (Small vs Big)
# Baseline: Majority class
majority_size_acc = max(np.mean(y_size_test == 1), np.mean(y_size_test == 0))
clf_lr = LogisticRegression(max_iter=1000, random_state=42)
clf_lr.fit(X_train, y_size_train)
lr_size_acc = accuracy_score(y_size_test, clf_lr.predict(X_test))

clf_rf = RandomForestClassifier(n_estimators=100, max_depth=5, random_state=42)
clf_rf.fit(X_train, y_size_train)
rf_size_acc = accuracy_score(y_size_test, clf_rf.predict(X_test))

print("\n--- ML OUT-OF-SAMPLE ACCURACY: SIZE (Small vs Big) ---")
print(f"  Random Guess / Baseline : 50.00% (Majority: {majority_size_acc*100:.2f}%)")
print(f"  Logistic Regression     : {lr_size_acc*100:.2f}%")
print(f"  Random Forest Classifier: {rf_size_acc*100:.2f}%")

# 2. Number Prediction (0-9)
clf_rf_num = RandomForestClassifier(n_estimators=100, max_depth=5, random_state=42)
clf_rf_num.fit(X_train, y_num_train)
rf_num_acc = accuracy_score(y_num_test, clf_rf_num.predict(X_test))

print("\n--- ML OUT-OF-SAMPLE ACCURACY: NUMBER (0-9) ---")
print(f"  Random Guess / Baseline : 10.00%")
print(f"  Random Forest Classifier: {rf_num_acc*100:.2f}%")

# 3. Colour Prediction
clf_rf_col = RandomForestClassifier(n_estimators=100, max_depth=5, random_state=42)
clf_rf_col.fit(X_train, y_col_train)
rf_col_acc = accuracy_score(y_col_test, clf_rf_col.predict(X_test))
majority_col_acc = np.max(np.bincount(pd.factorize(y_col_test)[0])) / len(y_col_test)

print("\n--- ML OUT-OF-SAMPLE ACCURACY: COLOUR ---")
print(f"  Random Guess / Baseline : ~40.00% (Majority: {majority_col_acc*100:.2f}%)")
print(f"  Random Forest Classifier: {rf_col_acc*100:.2f}%")
