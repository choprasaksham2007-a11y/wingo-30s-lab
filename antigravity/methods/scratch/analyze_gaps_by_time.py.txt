import openpyxl
from datetime import datetime, timedelta
from collections import defaultdict

wb = openpyxl.load_workbook(r"project_reports\Colour_Prediction_Dataset_v4.xlsx", data_only=True)
ws_gaps = wb['Missing_Rounds']

print("--- ALL 9 GAPS ---")
gaps = []
for r in range(2, ws_gaps.max_row + 1):
    g_idx = ws_gaps.cell(row=r, column=1).value
    p1 = ws_gaps.cell(row=r, column=2).value
    t1 = ws_gaps.cell(row=r, column=3).value
    p2 = ws_gaps.cell(row=r, column=4).value
    t2 = ws_gaps.cell(row=r, column=5).value
    missing = ws_gaps.cell(row=r, column=6).value
    dur = ws_gaps.cell(row=r, column=7).value
    gaps.append((g_idx, p1, t1, p2, t2, missing, dur))
    print(f"Gap {g_idx}: {missing} missing rounds | {dur} | From {t1} to {t2}")
