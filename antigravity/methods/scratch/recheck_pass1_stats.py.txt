import pandas as pd
import numpy as np
from scipy import stats
import math
import sys

sys.stdout.reconfigure(encoding='utf-8')

print("=" * 80)
print("PASS 1: STATISTICAL RANDOMNESS & AUTOCORRELATION BATTERY")
print("=" * 80)

df = pd.read_csv(r"project_reports\colour_raw_data_v18.csv", dtype={'Period ID': str})
N = len(df)
print(f"Total dataset size: N = {N:,} rounds")

# 1. Number Distribution & Chi-Square Goodness-of-Fit
obs_num = df['number'].value_counts().sort_index().values
exp_num = [N / 10.0] * 10
chi2_num, p_num = stats.chisquare(obs_num, exp_num)
print("\n1. Number Uniformity Test (Chi-Square Goodness-of-Fit):")
print(f"   Chi-Square Statistic = {chi2_num:.4f}, df = 9, p-value = {p_num:.5f}")
for i in range(10):
    print(f"   Digit {i}: Observed = {obs_num[i]:5d} ({obs_num[i]/N*100:5.2f}%), Expected = {N/10:.1f} (10.00%)")
if p_num > 0.05:
    print("   -> CONCLUSION: Fail to reject null hypothesis. Number distribution is UNIFORM and indistinguishable from pure randomness.")
else:
    print("   -> Statistically significant deviation detected.")

# 2. Size Uniformity Test (Small vs Big)
obs_sz = df['size'].value_counts()
exp_sz = [N / 2.0, N / 2.0]
chi2_sz, p_sz = stats.chisquare([obs_sz['Small'], obs_sz['Big']], exp_sz)
print(f"\n2. Size Distribution Test (Small vs Big):")
print(f"   Small: {obs_sz['Small']} ({obs_sz['Small']/N*100:.2f}%), Big: {obs_sz['Big']} ({obs_sz['Big']/N*100:.2f}%)")
print(f"   Chi-Square = {chi2_sz:.4f}, df = 1, p-value = {p_sz:.5f}")
if p_sz > 0.05:
    print("   -> CONCLUSION: Fail to reject null hypothesis. Size outcomes match exact 50/50 fair coin-flip.")

# 3. Colour Distribution Test
obs_col = df['colour'].value_counts()
# Pure Red: 40%, Pure Green: 40%, Red+Violet: 10%, Green+Violet: 10%
exp_col = [N * 0.40, N * 0.40, N * 0.10, N * 0.10]
actual_cols = [obs_col['Red'], obs_col['Green'], obs_col['Green+Violet'], obs_col['Red+Violet']]
chi2_col, p_col = stats.chisquare(actual_cols, exp_col)
print(f"\n3. Colour Distribution Test against Game Theoretical Probabilities:")
print(f"   Red (40% exp):          {obs_col['Red']} ({obs_col['Red']/N*100:.2f}%)")
print(f"   Green (40% exp):        {obs_col['Green']} ({obs_col['Green']/N*100:.2f}%)")
print(f"   Green+Violet (10% exp): {obs_col['Green+Violet']} ({obs_col['Green+Violet']/N*100:.2f}%)")
print(f"   Red+Violet (10% exp):   {obs_col['Red+Violet']} ({obs_col['Red+Violet']/N*100:.2f}%)")
print(f"   Chi-Square = {chi2_col:.4f}, df = 3, p-value = {p_col:.5f}")

# 4. Wald-Wolfowitz Runs Test for Random Sequence
def runs_test(series, name):
    # binary mapping
    if series.dtype == object or isinstance(series.iloc[0], str):
        vals = series.unique()
        b = (series == vals[0]).astype(int).values
    else:
        median_val = np.median(series)
        b = (series >= median_val).astype(int).values
    
    n1 = np.sum(b == 1)
    n0 = np.sum(b == 0)
    runs = 1 + np.sum(b[1:] != b[:-1])
    
    mu = 1.0 + (2.0 * n1 * n0) / (n1 + n0)
    var = (2.0 * n1 * n0 * (2.0 * n1 * n0 - n1 - n0)) / (((n1 + n0)**2) * (n1 + n0 - 1))
    z = (runs - mu) / math.sqrt(var)
    p_val = 2.0 * (1.0 - stats.norm.cdf(abs(z)))
    print(f"   {name}: Runs = {runs:,}, Expected = {mu:,.1f}, Z = {z:+.4f}, p-value = {p_val:.5f}")
    return z, p_val

print("\n4. Wald-Wolfowitz Runs Test (Clustering vs Alternation):")
runs_test(df['size'], "Size (Small vs Big)")
runs_test(df['number'], "Number (Above vs Below Median)")
runs_test(df['colour'].isin(['Green', 'Green+Violet']), "Green Parity (Green vs Red)")

# 5. Autocorrelation Analysis across Lags 1 to 30
print("\n5. Autocorrelation (ACF) Test (Lags 1 to 20):")
critical_val = 1.96 / math.sqrt(N)
print(f"   95% Confidence Threshold (No Memory): |r| < {critical_val:.4f}")

nums = df['number'].values
sizes = (df['size'] == 'Big').astype(int).values

significant_lags = []
for lag in range(1, 21):
    r_num = np.corrcoef(nums[:-lag], nums[lag:])[0, 1]
    r_sz = np.corrcoef(sizes[:-lag], sizes[lag:])[0, 1]
    star_num = " *" if abs(r_num) > critical_val else ""
    star_sz = " *" if abs(r_sz) > critical_val else ""
    if abs(r_num) > critical_val or abs(r_sz) > critical_val:
        significant_lags.append(lag)
    if lag <= 10 or lag % 5 == 0:
        print(f"   Lag {lag:2d}: r(number) = {r_num:+.4f}{star_num:2s} | r(size) = {r_sz:+.4f}{star_sz:2s}")

print(f"   Lags exceeding 95% threshold: {significant_lags} (Expected false alarm rate at alpha=0.05 is 5% = ~1 out of 20)")

# 6. Spectral FFT Peak Analysis (Periodic Cycle Detection)
print("\n6. Fast Fourier Transform (FFT) Periodicity Scan:")
fft_vals = np.abs(np.fft.rfft(nums - np.mean(nums)))[1:]
frequencies = np.fft.rfftfreq(len(nums))[1:]
periods = 1.0 / frequencies

max_fft_idx = np.argmax(fft_vals)
max_peak = fft_vals[max_fft_idx]
mean_fft = np.mean(fft_vals)
peak_ratio = max_peak / mean_fft
dominant_period = periods[max_fft_idx]

print(f"   Max FFT Peak / Mean Ratio: {peak_ratio:.2f}")
print(f"   Dominant Cycle Period (if any): {dominant_period:.2f} rounds")
if peak_ratio < 4.0:
    print("   -> CONCLUSION: No dominant cyclical peaks detected (flat white-noise spectrum).")
