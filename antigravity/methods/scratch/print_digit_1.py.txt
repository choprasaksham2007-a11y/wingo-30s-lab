import cv2, numpy as np

im = cv2.imread(r"extraction\extracted_zip_batch4\WhatsApp Image 2026-10-24 at 10.46.30 (1).jpeg")
# Let's find yc for 50172
# We can search around y in 500..1450 where g_mask has w=13
for yc in range(550, 1400, 10):
    crop = im[yc-20:yc+20, 315:375]
    hsv = cv2.cvtColor(crop, cv2.COLOR_BGR2HSV)
    g_mask = cv2.inRange(hsv, np.array([35, 50, 50]), np.array([85, 255, 255]))
    coords = np.argwhere(g_mask > 0)
    if len(coords) > 20:
        min_y, min_x = coords.min(axis=0)
        max_y, max_x = coords.max(axis=0)
        w = max_x - min_x + 1
        h = max_y - min_y + 1
        if w <= 14 and h >= 24:
            print(f"Found digit 1 at yc={yc}, w={w}, h={h}!")
            for r in g_mask[min_y:max_y+1, min_x:max_x+1]:
                print("".join(["#" if p > 0 else " " for p in r]))
            break
