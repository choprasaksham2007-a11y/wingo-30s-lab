def get_timestamp(pid: str) -> str:
    year = int(pid[:4])
    month = int(pid[4:6])
    day = int(pid[6:8])
    daily_round = int(pid[13:])
    base = datetime(year, month, day, 5, 30, 0)
    dt = base + timedelta(seconds=(daily_round - 1) * 30)
    return dt.strftime("%Y-%m-%d %H:%M:%S")

rows = []
for idx, r in df_src.iterrows():
    p = str(r["issueNumber"]).strip()
    prem = str(r["premium"]).strip() if pd.notna(r["premium"]) else ""
    num = int(r["number"])

    # Deterministic mappings
    sz = "Small" if num <= 4 else "Big"
    if num == 0:
        col = "Red+Violet"
    elif num == 5:
        col = "Green+Violet"
    elif num in [1, 3, 7, 9]:
        col = "Green"
    else:
        col = "Red"

    ts = get_timestamp(p)
    orig_col = str(r["colour"] if pd.notna(r["colour"]) else r.get("color", "")).strip()
    note = "Platsaas verified 5-digit premium" if len(prem) == 5 else "Legacy feed round"

    rows.append({
        "seq": idx + 1,
        "pid": p,
        "ts": ts,
        "prem": prem,
        "num": num,
        "col": col,
        "sz": sz,
        "orig_col": orig_col,
        "note": note
    })

print(f"Built {len(rows)} verified raw row records.")

# 2. Continuous Blocks and Gaps
def get_dt_from_pid(pid: str) -> datetime:
    year = int(pid[:4])
    month = int(pid[4:6])
    day = int(pid[6:8])
    daily_round = int(pid[13:])
    base = datetime(year, month, day, 5, 30, 0)
    return base + timedelta(seconds=(daily_round - 1) * 30)

blocks = []
cur_block = [rows[0]]

for i in range(1, len(rows)):
    dt_prev = get_dt_from_pid(rows[i-1]["pid"])
    dt_curr = get_dt_from_pid(rows[i]["pid"])
    if (dt_curr - dt_prev).total_seconds() == 30.0:
        cur_block.append(rows[i])
    else:
        blocks.append(cur_block)
        cur_block = [rows[i]]
blocks.append(cur_block)

print(f"Total Continuous Blocks: {len(blocks)}")

gaps = []
for i in range(len(blocks) - 1):
    b1 = blocks[i]
    b2 = blocks[i+1]
    p_last = b1[-1]["pid"]
    p_next = b2[0]["pid"]
    t_last = b1[-1]["ts"]
    t_next = b2[0]["ts"]

    dt1 = get_dt_from_pid(p_last)
    dt2 = get_dt_from_pid(p_next)
    missing = int((dt2 - dt1).total_seconds() / 30.0) - 1

    sec = missing * 30
    h_d = sec // 3600
    m_d = (sec % 3600) // 60
    s_d = sec % 60
    dur_str = f"{h_d}:{m_d:02d}:{s_d:02d}"

    gaps.append({
        "gap_idx": i + 1,
        "prec_pid": p_last,
        "prec_time": t_last,
        "res_pid": p_next,
        "res_time": t_next,
        "missing_rounds": missing,
        "duration": dur_str,
        "block_before": i + 1,
        "block_after": i + 2
    })

print(f"Total Gaps documented: {len(gaps)}")

# 3. Generate ML_Ready Data (Lag 5 order, strictly inside blocks, zero gap crossings)

            sample_idx,

            tgt["ts"],

            tgt["num"],
            tgt["col"],
            tgt["sz"],
            p1["num"], p1["col"], p1["sz"],
            p2["num"], p2["col"], p2["sz"],
            p3["num"], p3["col"], p3["sz"],
            p4["num"], p4["col"], p4["sz"],
            p5["num"], p5["col"], p5["sz"],
            b_idx
        ]
        ml_rows_list.append(ml_row)
        sample_idx += 1

print(f"Total ML Training Samples generated: {len(ml_rows_list)} (0 gap crossings across {len(blocks)} blocks)")

def get_sha256(path: str) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest()
