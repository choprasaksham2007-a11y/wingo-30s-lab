import cv2, numpy as np

crop = cv2.imread(r"scratch\crop_50148.png")
hsv = cv2.cvtColor(crop, cv2.COLOR_BGR2HSV)
g_mask = cv2.inRange(hsv, np.array([35, 50, 50]), np.array([85, 255, 255]))

# Let's print out the ASCII art of g_mask!
for row in g_mask:
    if np.any(row):
        print("".join(["#" if p > 0 else " " for p in row]))
