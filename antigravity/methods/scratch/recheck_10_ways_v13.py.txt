# WAY 1: DETERMINISTIC NUMBER-TO-COLOUR RULE DETERMINISM
# -----------------------------------------------------------------------------
print("--> WAY 1: Deterministic Colour Mapping Rule Check (100% of rows)...")
way1_errors = []
for r in range(2, ws_raw.max_row + 1):
    seq = ws_raw.cell(row=r, column=1).value
    pid = str(ws_raw.cell(row=r, column=2).value)
    num = ws_raw.cell(row=r, column=4).value
    col = str(ws_raw.cell(row=r, column=5).value)
    
    if num == 0: exp_col = "Red+Violet"
    elif num == 5: exp_col = "Green+Violet"
    elif num in [1, 3, 7, 9]: exp_col = "Green"
    elif num in [2, 4, 6, 8]: exp_col = "Red"
    else: exp_col = "INVALID"
    
    if col != exp_col:
        way1_errors.append((seq, pid, num, col, exp_col))

assert len(way1_errors) == 0, f"Way 1 Failed: {len(way1_errors)} errors"
print(f"    [PASS] Way 1: 0 errors across {total_raw_rows} rows. 100% deterministic colour compliance.\n")

# -----------------------------------------------------------------------------
# WAY 2: DETERMINISTIC NUMBER-TO-SIZE RULE DETERMINISM

print("--> WAY 2: Deterministic Size Mapping Rule Check (100% of rows)...")

# -----------------------------------------------------------------------------
# WAY 5: INTRA-BLOCK MICROSECOND-LEVEL CADENCE INVARIANCE CHECK
# -----------------------------------------------------------------------------
print("--> WAY 5: Intra-Block 30.000s Cadence Precision Check...")
timestamps = [datetime.strptime(str(ws_raw.cell(row=r, column=3).value), "%Y-%m-%d %H:%M:%S") for r in range(2, ws_raw.max_row + 1)]

blocks = []
cur_block = [0]
for i in range(1, len(raw_pids)):
    p1, p2 = raw_pids[i-1], raw_pids[i]
    d1, d2 = p1[:8], p2[:8]
    r1, r2 = int(p1[-5:]) - 50000, int(p2[-5:]) - 50000
    
    is_step = False
    if d1 == d2 and r2 == r1 + 1:
        is_step = True
    elif r1 == 2880 and r2 == 1:
        is_step = True
        
    if is_step:
        cur_block.append(i)
    else:
        blocks.append(cur_block)
        cur_block = [i]
blocks.append(cur_block)

way5_cadence_errors = []

for b_idx, b in enumerate(blocks, 1):

        dt = (timestamps[i2] - timestamps[i1]).total_seconds()
        intra_steps += 1
        if dt != 30.0:
            way5_cadence_errors.append((i1+1, i2+1, raw_pids[i1], raw_pids[i2], dt))

# WAY 10: MULTI-DIRECTORY CRYPTOGRAPHIC SHA-256 HASH INVARIANCE CHECK

print("--> WAY 10: Multi-Directory SHA-256 Hash Invariance Check...")
def get_sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest()

files_to_check = [
    "Colour_Prediction_Dataset_v13.xlsx",
    "colour_raw_data_v13.csv",
    "colour_ml_ready_v13.csv"
]

    p_d = os.path.join(dl_dir, fname)

    assert os.path.exists(p_d), f"Missing in Downloads: {fname}"
    
    sb, sd = os.path.getsize(p_b), os.path.getsize(p_d)
    assert sb == sd, f"Size mismatch for {fname}: {sb} vs {sd}"
    
    hb, hd = get_sha256(p_b), get_sha256(p_d)
    assert hb == hd, f"Hash mismatch for {fname}: {hb} vs {hd}"
    
    print(f"    [PASS] {fname}: {sb:,} bytes | SHA-256: {hb}")
