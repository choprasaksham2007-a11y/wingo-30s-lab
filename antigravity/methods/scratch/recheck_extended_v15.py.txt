wb = openpyxl.load_workbook(excel_path, data_only=True)

print(f"Loaded Master v15: {raw_rows} raw rows, {ml_rows} ML rows, {gap_rows} gaps.\n")

# -----------------------------------------------------------------------------
# PASS 1: TAIL INTEGRITY & USER UPLOADED CSV END-TO-END MATCH
# -----------------------------------------------------------------------------
print("--> PASS 1: Tail Integrity & User CSV Verification...")
user_rows = []
with open(user_csv_path, 'r', encoding='utf-8') as f:
    reader = csv.DictReader(f)
    for r in reader:
        user_rows.append(r)

print(f"    User feed row count: {len(user_rows):,} rows.")
first_user_pid = str(user_rows[0].get('issueNumber') or user_rows[0].get('Period ID') or user_rows[0].get('period'))
last_user_pid = str(user_rows[-1].get('issueNumber') or user_rows[-1].get('Period ID') or user_rows[-1].get('period'))
print(f"    First user feed PID: {first_user_pid}")
print(f"    Last user feed PID:  {last_user_pid}")

last_excel_pid = str(ws_raw.cell(row=raw_rows+1, column=2).value)
last_excel_num = ws_raw.cell(row=raw_rows+1, column=4).value
last_excel_col = ws_raw.cell(row=raw_rows+1, column=5).value
last_excel_sz = ws_raw.cell(row=raw_rows+1, column=6).value
last_excel_time = str(ws_raw.cell(row=raw_rows+1, column=3).value)

assert last_excel_pid == last_user_pid, f"Tail mismatch: {last_excel_pid} vs {last_user_pid}"
print(f"    [PASS] Tail round strictly matches user upload: {last_excel_pid} ({last_excel_time})")
print(f"           Number: {last_excel_num}, Colour: {last_excel_col}, Size: {last_excel_sz}\n")

# -----------------------------------------------------------------------------
# PASS 2: DE-DUPLICATION MATHEMATICAL PROOF (v14 vs Incoming Feed vs v15)
# -----------------------------------------------------------------------------
print("--> PASS 2: De-Duplication Mathematical Proof...")
v14_pids = set()
with open(v14_csv_path, 'r', encoding='utf-8') as f:
    reader = csv.DictReader(f)
    for r in reader:
        v14_pids.add(r['Period ID'])
print(f"    v14 Master row count: {len(v14_pids):,} rows.")

incoming_pids = [str(r.get('issueNumber') or r.get('Period ID') or r.get('period')) for r in user_rows]
incoming_unique = set(incoming_pids)
print(f"    Incoming feed unique PIDs: {len(incoming_unique):,} (Total rows: {len(incoming_pids):,})")

overlap_pids = incoming_unique.intersection(v14_pids)
new_pids = incoming_unique - v14_pids

print(f"    Calculated Overlap: {len(overlap_pids):,} rows.")
print(f"    Calculated Truly New: {len(new_pids):,} rows.")
assert len(overlap_pids) == 2048, f"Expected 2048 overlap, got {len(overlap_pids)}"
assert len(new_pids) == 1743, f"Expected 1743 new, got {len(new_pids)}"

# Now verify v15 contains exactly v14 + new_pids, and zero duplicates
v15_raw_pids = [str(ws_raw.cell(row=r, column=2).value) for r in range(2, ws_raw.max_row + 1)]
assert len(v15_raw_pids) == len(v14_pids) + len(new_pids), "v15 count != v14 + new_pids"
assert len(v15_raw_pids) == len(set(v15_raw_pids)), "Duplicates detected in v15!"
assert set(v15_raw_pids) == v14_pids.union(new_pids), "v15 set != v14 union new_pids"
print(f"    [PASS] Mathematical proof verified: v15 = v14 ({len(v14_pids)}) + new ({len(new_pids)}) = {len(v15_raw_pids)}.")
print("           Zero overlapping rows were duplicated. Exactly 0 redundant rows copied.\n")

# -----------------------------------------------------------------------------
# PASS 3: DETERMINISTIC RULES EXHAUSTIVE VALIDATION
# -----------------------------------------------------------------------------
print("--> PASS 3: Deterministic Colour & Size Rules (Every Single Row)...")
num_counts = {i: 0 for i in range(10)}
col_counts = {"Green": 0, "Red": 0, "Green+Violet": 0, "Red+Violet": 0}
sz_counts = {"Small": 0, "Big": 0}

for r in range(2, ws_raw.max_row + 1):
    n = ws_raw.cell(row=r, column=4).value
    c = str(ws_raw.cell(row=r, column=5).value)
    s = str(ws_raw.cell(row=r, column=6).value)
    
    assert isinstance(n, int) and 0 <= n <= 9, f"Invalid number at row {r}: {n}"
    num_counts[n] += 1
    col_counts[c] += 1
    sz_counts[s] += 1
    
    # Deterministic checks
    if n == 0: assert c == "Red+Violet"
    elif n == 5: assert c == "Green+Violet"
    elif n in [1, 3, 7, 9]: assert c == "Green"
    elif n in [2, 4, 6, 8]: assert c == "Red"

# -----------------------------------------------------------------------------
# PASS 4: CADENCE, BLOCKS, AND GAPS MATHEMATICAL BIJECTION
# -----------------------------------------------------------------------------
print("--> PASS 4: Cadence, Blocks, and Gaps Bijection...")
raw_timestamps = [datetime.strptime(str(ws_raw.cell(row=r, column=3).value), "%Y-%m-%d %H:%M:%S") for r in range(2, ws_raw.max_row + 1)]

detected_blocks = []
current_block = [0]
for i in range(1, len(v15_raw_pids)):
    p1, p2 = v15_raw_pids[i-1], v15_raw_pids[i]
    d1, d2 = p1[:8], p2[:8]
    r1, r2 = int(p1[-5:]) - 50000, int(p2[-5:]) - 50000
    
    is_step = False
    if d1 == d2 and r2 == r1 + 1:
        is_step = True
    elif r1 == 2880 and r2 == 1:
        is_step = True
        
    if is_step:
        current_block.append(i)
    else:
        detected_blocks.append(current_block)
        current_block = [i]
detected_blocks.append(current_block)

assert len(detected_blocks) == 44, f"Expected 44 blocks, found {len(detected_blocks)}"
assert len(detected_blocks) - 1 == gap_rows, f"Gap count mismatch: {gap_rows} vs {len(detected_blocks)-1}"

# Verify every intra-block step is exactly 30s
intra_steps = 0
for b in detected_blocks:
    for j in range(len(b) - 1):
        dt = (raw_timestamps[b[j+1]] - raw_timestamps[b[j]]).total_seconds()
        assert dt == 30.0, f"Drift at {v15_raw_pids[b[j]]}: {dt}s"
        intra_steps += 1
assert intra_steps == raw_rows - 44
print(f"    [PASS] All {len(detected_blocks)} blocks verified. All {intra_steps} intra-block steps have exactly 30.000s cadence (0.000s drift).")

# Verify all gaps match detected gaps exactly
total_missing_rounds = 0
for g_idx in range(len(detected_blocks) - 1):
    b_curr = detected_blocks[g_idx]
    b_next = detected_blocks[g_idx + 1]
    
    p_last = v15_raw_pids[b_curr[-1]]
    t_last = raw_timestamps[b_curr[-1]]
    p_first = v15_raw_pids[b_next[0]]
    t_first = raw_timestamps[b_next[0]]
    
    excel_g_row = g_idx + 2
    eg_id = ws_gaps.cell(row=excel_g_row, column=1).value
    eg_ppid = str(ws_gaps.cell(row=excel_g_row, column=2).value)
    eg_ptime = datetime.strptime(str(ws_gaps.cell(row=excel_g_row, column=3).value), "%Y-%m-%d %H:%M:%S")
    eg_rpid = str(ws_gaps.cell(row=excel_g_row, column=4).value)
    eg_rtime = datetime.strptime(str(ws_gaps.cell(row=excel_g_row, column=5).value), "%Y-%m-%d %H:%M:%S")
    eg_miss = int(ws_gaps.cell(row=excel_g_row, column=6).value)
    
    assert eg_id == g_idx + 1
    assert eg_ppid == p_last
    assert eg_ptime == t_last
    assert eg_rpid == p_first
    assert eg_rtime == t_first
    
    dt = (t_first - t_last).total_seconds()
    exp_miss = int(round(dt / 30.0)) - 1
    assert eg_miss == exp_miss
    total_missing_rounds += eg_miss

assert total_missing_rounds == 7906
print(f"    [PASS] All {gap_rows} gaps verified with 100% mathematical bijection. Total missing rounds: {total_missing_rounds:,}.\n")

def get_hash(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest(), os.path.getsize(path)

files_to_check = [
    "Colour_Prediction_Dataset_v15.xlsx",
    "colour_raw_data_v15.csv",
    "colour_ml_ready_v15.csv",
    "audit_certification_report_v15.md"
]

for fname in files_to_check:

    dp = os.path.join(dl_dir, fname)
    bh, bs = get_hash(bp)
    dh, ds = get_hash(dp)
    assert bh == dh, f"Hash mismatch for {fname}"
    assert bs == ds, f"Size mismatch for {fname}"
    print(f"    [PASS] {fname}: {bs:,} bytes | SHA-256: {bh}")
