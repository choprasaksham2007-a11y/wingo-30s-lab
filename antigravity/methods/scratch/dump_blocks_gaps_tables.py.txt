import openpyxl

wb = openpyxl.load_workbook(r"project_reports\Colour_Prediction_Dataset_v18.xlsx", read_only=True)
ws_raw = wb['Raw_Data']
ws_gaps = wb['Missing_Rounds']

# Collect block sequence ranges
raw_rows = list(ws_raw.iter_rows(min_row=2, values_only=True))
gaps_rows = list(ws_gaps.iter_rows(min_row=2, values_only=True))

# Identify blocks
from datetime import datetime, timedelta
def get_dt(pid):
    year = int(pid[:4])
    month = int(pid[4:6])
    day = int(pid[6:8])
    suffix = int(pid[-5:])
    base = datetime(year, month, day, 5, 30, 0)
    return base + timedelta(seconds=(suffix - 50000) * 30)

blocks = []
cur_block = [raw_rows[0]]
for i in range(1, len(raw_rows)):
    t1 = get_dt(str(raw_rows[i-1][1]))
    t2 = get_dt(str(raw_rows[i][1]))
    if (t2 - t1).total_seconds() == 30.0:
        cur_block.append(raw_rows[i])
    else:
        blocks.append(cur_block)
        cur_block = [raw_rows[i]]
blocks.append(cur_block)

print("\nBlocks 58 to 73:")
print("| Block ID | Start Sequence | End Sequence | Round Count | Start Period ID | End Period ID | Start Time (IST) | End Time (IST) |")
print("| :---: | :---: | :---: | :---: | :---: | :---: | :---: | :---: |")
for b_idx in range(58, len(blocks) + 1):
    b = blocks[b_idx - 1]
    s_seq = b[0][0]
    e_seq = b[-1][0]
    cnt = len(b)
    s_pid = b[0][1]
    e_pid = b[-1][1]
    s_t = b[0][2]
    e_t = b[-1][2]
    print(f"| **Block {b_idx}** | {s_seq:,} | {e_seq:,} | {cnt} | `{s_pid}` | `{e_pid}` | {s_t} | {e_t} |")

print("\nGaps 58 to 72:")
print("| Gap # | Preceding PID | Preceding Time (IST) | Resuming PID | Resuming Time (IST) | Missing Rounds | Duration |")
print("| :---: | :---: | :---: | :---: | :---: | :---: | :---: |")
for g in gaps_rows[57:]:
    g_num = g[0]
    p1 = g[1]
    t1 = g[2]
    p2 = g[3]
    t2 = g[4]
    m = g[5]
    d = g[6]
    print(f"| **Gap {g_num}** | `{p1}` | {t1} | `{p2}` | {t2} | {m:,} | {d} |")
