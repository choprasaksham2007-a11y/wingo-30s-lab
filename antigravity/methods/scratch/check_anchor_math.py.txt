from datetime import datetime, timedelta

# Let's check Anchor 1 on 2026-10-23:
# PID: 20261023100050494 -> 2026-10-23 09:37:00
a1_dt = datetime(2026, 10, 23, 9, 37, 0)
a1_suffix = 50494

# Round 50001 on 2026-10-23:
rounds_from_50001 = a1_suffix - 50001 # 493 rounds
round_50001_time = a1_dt - timedelta(seconds=rounds_from_50001 * 30)
print(f"Round 50001 on 2026-10-23 time: {round_50001_time}")

# What was Anchor 2:
# 20261023100052278 -> 2026-10-24 00:29:00
a2_dt = datetime(2026, 10, 24, 0, 29, 0)
a2_suffix = 52278
print("A2 diff from A1:", (a2_dt - a1_dt).total_seconds(), "seconds")
print("A2 rounds diff:", (a2_suffix - a1_suffix) * 30, "seconds")
print("Match exactly:", (a2_dt - a1_dt).total_seconds() == (a2_suffix - a1_suffix) * 30)

# Now what about 2026-10-24?
# In this game, does the day reset at round 52880 -> 50001?
# There are 2880 rounds per day: 2880 * 30s = 86400s = 24 hours!
# Round 52880 is the 2880th round of 2026-10-23.
# Round 52880 time = round_50001_time + (2880 - 1) * 30s
r_last_day1 = round_50001_time + timedelta(seconds=(2880 - 1) * 30)
print(f"Last round of 2026-10-23 (52880): {r_last_day1}")
# Next round is 50001 of 2026-10-24:
r_first_day2 = r_last_day1 + timedelta(seconds=30)
print(f"First round of 2026-10-24 (50001): {r_first_day2}")

# Now let's calculate round 50638 on 2026-10-24:
r_50638_time = r_first_day2 + timedelta(seconds=(50638 - 50001) * 30)
print(f"Calculated time for 20261024100050638: {r_50638_time}")
