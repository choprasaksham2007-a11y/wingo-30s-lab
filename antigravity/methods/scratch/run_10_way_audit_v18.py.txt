# WAY 6: Exact 30.000s Intra-Block Cadence (0.000s Drift)
p("\n[WAY 6] Verifying Exact 30.000s Intra-Block Cadence...")
def get_dt(pid):
    year = int(pid[:4])
    month = int(pid[4:6])
    day = int(pid[6:8])
    suffix = int(pid[-5:])
    base = datetime(year, month, day, 5, 30, 0)
    return base + timedelta(seconds=(suffix - 50000) * 30)

blocks = []
cur_block = [raw_rows[0]]
for i in range(1, len(raw_rows)):
    t_prev = get_dt(raw_rows[i-1]['pid'])
    t_curr = get_dt(raw_rows[i]['pid'])
    if (t_curr - t_prev).total_seconds() == 30.0:
        cur_block.append(raw_rows[i])
    else:
        blocks.append(cur_block)
        cur_block = [raw_rows[i]]
blocks.append(cur_block)

block_drift_violations = []
for b_idx, block in enumerate(blocks, 1):
    for i in range(1, len(block)):

        dt2 = datetime.strptime(block[i]['ts'], '%Y-%m-%d %H:%M:%S')
        diff = (dt2 - dt1).total_seconds()
        if diff != 30.0:
            block_drift_violations.append((b_idx, block[i-1]['pid'], block[i]['pid'], diff))

if len(block_drift_violations) == 0 and len(blocks) == 73:

p("\n[WAY 10] Verifying Bitwise SHA-256 Multi-Directory Synchronization...")
def get_sha256(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest()

hash_excel_b = get_sha256(excel_brain)
hash_excel_d = get_sha256(excel_down)
hash_raw_b = get_sha256(raw_csv_brain)
hash_raw_d = get_sha256(raw_csv_down)
hash_ml_b = get_sha256(ml_csv_brain)
