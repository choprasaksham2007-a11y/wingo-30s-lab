import os, sys, json, openpyxl
from collections import Counter
from openpyxl.styles import PatternFill, Font, Alignment, Border, Side
from datetime import datetime, timedelta

# Paths
brain_dir = r"project_reports"
v4_path = os.path.join(brain_dir, "Colour_Prediction_Dataset_v4.xlsx")
v3_path = os.path.join(brain_dir, "Colour_Prediction_Dataset_v3.xlsx")

# Load existing v3 data from Raw_Data
wb_v3 = openpyxl.load_workbook(v3_path, data_only=True)
ws_v3 = wb_v3['Raw_Data']

v3_records = []
for r in range(2, ws_v3.max_row + 1):
    pid = str(ws_v3.cell(row=r, column=2).value)
    ts = str(ws_v3.cell(row=r, column=3).value)
    num = int(ws_v3.cell(row=r, column=4).value)
    col = str(ws_v3.cell(row=r, column=5).value)
    sz = str(ws_v3.cell(row=r, column=6).value)
    col_orig = str(ws_v3.cell(row=r, column=7).value or col)
    note = str(ws_v3.cell(row=r, column=8).value or "Original extraction verified")
    v3_records.append({
        "period_id": pid,
        "timestamp": ts,
        "number": num,
        "colour": col,
        "size": sz,
        "colour_original": col_orig,
        "fix_note": note
    })

# Load Batch 4
with open(r"extraction\reconciled_batch4_calibrated.json", "r") as f:
    b4_data = json.load(f)

# Combine
all_records = list(v3_records)
existing_pids = set(r["period_id"] for r in v3_records)

for pid in sorted(b4_data.keys()):
    if pid not in existing_pids:
        item = b4_data[pid]
        all_records.append({
            "period_id": pid,
            "timestamp": item["timestamp"],
            "number": item["number"],
            "colour": item["colour"],
            "size": item["size"],
            "colour_original": item["colour"],
            "fix_note": "Batch 4 extraction verified"
        })

all_records.sort(key=lambda r: int(r["period_id"]))
print(f"Total merged records: {len(all_records)}")

# Re-number sequence
for idx, r in enumerate(all_records, 1):
    r["sequence_no"] = idx

# Consecutive checker
def is_consecutive(pid1, pid2):
    p1 = int(pid1)
    p2 = int(pid2)
    s1 = int(pid1[-5:])
    s2 = int(pid2[-5:])
    if p2 == p1 + 1 and s2 == s1 + 1:
        return True
    if s1 == 52880 and s2 == 50001:
        d1 = datetime.strptime(pid1[:8], "%Y%m%d")
        d2 = datetime.strptime(pid2[:8], "%Y%m%d")
        if (d2 - d1).days == 1:
            return True
    return False

# Blocks & Gaps
blocks = []
gaps = []
curr_block = [all_records[0]]

for i in range(1, len(all_records)):
    p_rec = all_records[i-1]
    c_rec = all_records[i]
    if is_consecutive(p_rec["period_id"], c_rec["period_id"]):
        curr_block.append(c_rec)
    else:
        blocks.append(curr_block)
        p1 = p_rec["period_id"]
        p2 = c_rec["period_id"]
        t1 = datetime.strptime(p_rec["timestamp"], "%Y-%m-%d %H:%M:%S")
        t2 = datetime.strptime(c_rec["timestamp"], "%Y-%m-%d %H:%M:%S")
        sec_diff = int((t2 - t1).total_seconds())
        missing_count = (sec_diff // 30) - 1
        gaps.append({
            "gap_index": len(gaps) + 1,
            "from_pid": p1,
            "to_pid": p2,
            "from_time": p_rec["timestamp"],
            "to_time": c_rec["timestamp"],
            "missing_rounds": missing_count,
            "duration_sec": sec_diff,
            "block_before": len(blocks),
            "block_after": len(blocks) + 1
        })
        curr_block = [c_rec]
blocks.append(curr_block)

print(f"Total Blocks: {len(blocks)}, Total Gaps: {len(gaps)}")

# ML_Ready samples
ml_samples = []
for b_idx, block in enumerate(blocks, 1):
    if len(block) <= 5:
        continue
    for i in range(5, len(block)):
        target = block[i]
        sample = {
            "target_period_id": target["period_id"],
            "target_timestamp": target["timestamp"],
            "target_number": target["number"],
            "target_colour": target["colour"],
            "target_size": target["size"],
            "block_id": b_idx
        }
        for lag in range(1, 6):
            p = block[i - lag]
            sample[f"prev_{lag}_number"] = p["number"]
            sample[f"prev_{lag}_colour"] = p["colour"]
            sample[f"prev_{lag}_size"] = p["size"]
        ml_samples.append(sample)

print(f"Total ML samples: {len(ml_samples)} (Expected: {len(all_records) - 5 * len(blocks)})")

# Styles
navy_fill = PatternFill(start_color="1F497D", end_color="1F497D", fill_type="solid")
header_font = Font(name="Calibri", size=11, bold=True, color="FFFFFF")
regular_font = Font(name="Calibri", size=11, bold=False, color="000000")
bold_font = Font(name="Calibri", size=11, bold=True, color="000000")

left_align = Alignment(horizontal="left", vertical="center")
thin_border = Border(
    left=Side(style='thin', color='D9D9D9'),
    right=Side(style='thin', color='D9D9D9'),
    top=Side(style='thin', color='D9D9D9'),
    bottom=Side(style='thin', color='D9D9D9')
)

wb = openpyxl.Workbook()
wb.remove(wb.active) # Remove default sheet

# 1. Raw_Data
ws_raw = wb.create_sheet(title="Raw_Data")
raw_headers = ["sequence_no", "Period ID", "timestamp", "number", "colour", "size", "colour_original", "fix_note"]
ws_raw.append(raw_headers)
for col_num in range(1, len(raw_headers) + 1):
    c = ws_raw.cell(row=1, column=col_num)
    c.fill = navy_fill
    c.font = header_font
    c.alignment = center_align

for r_idx, r in enumerate(all_records, 2):
    ws_raw.cell(row=r_idx, column=1, value=r["sequence_no"]).alignment = center_align
    c_pid = ws_raw.cell(row=r_idx, column=2, value=str(r["period_id"]))
    c_pid.number_format = '@'
    c_pid.alignment = center_align
    ws_raw.cell(row=r_idx, column=3, value=r["timestamp"]).alignment = center_align
    ws_raw.cell(row=r_idx, column=4, value=r["number"]).alignment = center_align
    ws_raw.cell(row=r_idx, column=5, value=r["colour"]).alignment = center_align
    ws_raw.cell(row=r_idx, column=6, value=r["size"]).alignment = center_align
    ws_raw.cell(row=r_idx, column=7, value=r["colour_original"]).alignment = center_align
    ws_raw.cell(row=r_idx, column=8, value=r["fix_note"]).alignment = left_align

# 2. ML_Ready
ws_ml = wb.create_sheet(title="ML_Ready")
ml_headers = [
    "target_period_id", "target_timestamp", "target_number", "target_colour", "target_size",
    "prev_1_number", "prev_1_colour", "prev_1_size",
    "prev_2_number", "prev_2_colour", "prev_2_size",

ws_ml.append(ml_headers)
for col_num in range(1, len(ml_headers) + 1):
    c = ws_ml.cell(row=1, column=col_num)

    c.font = header_font
    c.alignment = center_align

for r_idx, s in enumerate(ml_samples, 2):
    c_pid = ws_ml.cell(row=r_idx, column=1, value=str(s["target_period_id"]))

    c_pid.alignment = center_align
    ws_ml.cell(row=r_idx, column=2, value=s["target_timestamp"]).alignment = center_align

    ws_ml.cell(row=r_idx, column=4, value=s["target_colour"]).alignment = center_align
    ws_ml.cell(row=r_idx, column=5, value=s["target_size"]).alignment = center_align

        ws_ml.cell(row=r_idx, column=col_c, value=s[f"prev_{lag}_number"]).alignment = center_align
        ws_ml.cell(row=r_idx, column=col_c+1, value=s[f"prev_{lag}_colour"]).alignment = center_align
        ws_ml.cell(row=r_idx, column=col_c+2, value=s[f"prev_{lag}_size"]).alignment = center_align
        col_c += 3

    c.fill = navy_fill
    c.font = header_font
    c.alignment = center_align

for r_idx, g in enumerate(gaps, 2):

    src_ws = wb_v3[sname]
    new_ws = wb.create_sheet(title=sname)
    for row in src_ws.iter_rows(values_only=False):
        new_row = []
        for cell in row:
            new_row.append(cell.value)
        new_ws.append(new_row)
        for col_idx, cell in enumerate(row, 1):
            target_cell = new_ws.cell(row=new_ws.max_row, column=col_idx)
            if cell.number_format == '@':
                target_cell.number_format = '@'
            if cell.font and cell.font.bold:
                target_cell.font = bold_font

# Auto-adjust column widths
for ws in [ws_raw, ws_ml, ws_gaps]:
    for col in ws.columns:
        max_len = max(len(str(cell.value or '')) for cell in col)
        col_letter = openpyxl.utils.get_column_letter(col[0].column)
        ws.column_dimensions[col_letter].width = max(max_len + 3, 12)
