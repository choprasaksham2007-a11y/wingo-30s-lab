import openpyxl
import pandas as pd
import numpy as np

excel_path = r"project_reports\Colour_Prediction_Dataset_v8.xlsx"
csv_path = r"project_reports\colour_raw_data_v8.csv"

print("=== AUDIT PART 1: DETERMINISTIC GAME RULES AUDIT (3,425 ROWS) ===")

# Load Excel sheet Raw_Data
df_excel = pd.read_excel(excel_path, sheet_name='Raw_Data')
df_csv = pd.read_csv(csv_path)

print(f"Excel Raw_Data rows: {len(df_excel)}")
print(f"CSV rows:            {len(df_csv)}")

assert len(df_excel) == 3425, f"Expected 3425 rows, found {len(df_excel)}"
assert len(df_csv) == 3425, f"Expected 3425 rows in CSV, found {len(df_csv)}"

# Verify CSV and Excel Raw_Data match exactly
for col in ['sequence_no', 'Period ID', 'timestamp', 'number', 'colour', 'size']:
    excel_col = df_excel[col].astype(str)
    csv_col = df_csv[col].astype(str)
    mismatches = (excel_col != csv_col).sum()
    print(f"Column '{col}' Excel vs CSV mismatches: {mismatches}")
    assert mismatches == 0, f"Mismatch in column {col}"

# Deterministic Game Rules:
# 0 -> Red+Violet, Small
# 5 -> Green+Violet, Big
# 1, 3 -> Green, Small
# 7, 9 -> Green, Big
# 2, 4 -> Red, Small
# 6, 8 -> Red, Big

expected_rules = {
    0: ('Red+Violet', 'Small'),
    1: ('Green', 'Small'),
    2: ('Red', 'Small'),
    3: ('Green', 'Small'),
    4: ('Red', 'Small'),
    5: ('Green+Violet', 'Big'),
    6: ('Red', 'Big'),
    7: ('Green', 'Big'),
    8: ('Red', 'Big'),
    9: ('Green', 'Big'),
}

violations = []
for idx, row in df_excel.iterrows():
    seq = row['sequence_no']
    pid = str(row['Period ID'])
    num = row['number']
    col = row['colour']
    sz = row['size']
    
    if num not in expected_rules:
        violations.append((seq, pid, num, col, sz, f"Invalid number {num}"))
        continue
    
    exp_col, exp_sz = expected_rules[num]
    if col != exp_col:
        violations.append((seq, pid, num, col, sz, f"Colour mismatch: expected {exp_col}, got {col}"))
    if sz != exp_sz:
        violations.append((seq, pid, num, col, sz, f"Size mismatch: expected {exp_sz}, got {sz}"))

print(f"\nRule Violations Count across all 3,425 rows: {len(violations)}")
if violations:
    for v in violations[:10]:
        print("  Violation:", v)
else:
    print(">> RULE VIOLATION COUNT IS EXACTLY 0. PERFECT COMPLIANCE! <<")

# Sequence Continuity Check
seq_diffs = np.diff(df_excel['sequence_no'])
seq_gaps = np.where(seq_diffs != 1)[0]
print(f"\nSequence continuity errors (gaps or duplicate sequence_no): {len(seq_gaps)}")
assert len(seq_gaps) == 0, f"Sequence continuity errors found: {seq_gaps}"
assert df_excel['sequence_no'].iloc[0] == 1, "First sequence_no is not 1"
assert df_excel['sequence_no'].iloc[-1] == 3425, "Last sequence_no is not 3425"

# Period ID Format Integrity
pids = df_excel['Period ID'].astype(str)
invalid_pids = []
for pid in pids:
    if len(pid) != 17 or not pid.isdigit():
        invalid_pids.append(pid)
print(f"Period ID format errors (must be 17-digit numeric string): {len(invalid_pids)}")
assert len(invalid_pids) == 0

# Check Excel cell data type for Period ID (should be string, no float notation)
wb = openpyxl.load_workbook(excel_path, data_only=False)
ws = wb['Raw_Data']
col_idx = None
for c in range(1, ws.max_column + 1):
    if ws.cell(row=1, column=c).value == 'Period ID':
        col_idx = c
        break
print(f"Period ID is column {col_idx}")

non_string_cells = []
for r in range(2, ws.max_row + 1):
    cell = ws.cell(row=r, column=col_idx)
    val = cell.value
    # Check if value is string and format is '@' or text
    if not isinstance(val, str) or len(val) != 17:
        non_string_cells.append((r, val, type(val), cell.number_format))

print(f"Period ID non-string or corrupt cells in Excel: {len(non_string_cells)}")
assert len(non_string_cells) == 0, f"Corrupt cells found: {non_string_cells[:5]}"

print("\n=== RULE COMPLIANCE AUDIT PASSED 100% ===")
