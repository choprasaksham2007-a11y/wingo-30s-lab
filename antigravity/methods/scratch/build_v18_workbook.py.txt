def get_timestamp(pid: str) -> str:
    year = int(pid[:4])
    month = int(pid[4:6])
    day = int(pid[6:8])
    suffix = int(pid[-5:])
    base = datetime(year, month, day, 5, 30, 0)
    dt = base + timedelta(seconds=(suffix - 50000) * 30)
    return dt.strftime('%Y-%m-%d %H:%M:%S')

# 3. Calculate Blocks and Gaps
def get_dt_from_pid(pid: str) -> datetime:
    year = int(pid[:4])
    month = int(pid[4:6])
    day = int(pid[6:8])
    suffix = int(pid[-5:])
    base = datetime(year, month, day, 5, 30, 0)
    return base + timedelta(seconds=(suffix - 50000) * 30)

blocks = []
cur_block = [rows[0]]

for i in range(1, len(rows)):
    dt_prev = get_dt_from_pid(rows[i-1]['pid'])
    dt_curr = get_dt_from_pid(rows[i]['pid'])
    
    if (dt_curr - dt_prev).total_seconds() == 30.0:
        cur_block.append(rows[i])
    else:
        blocks.append(cur_block)
        cur_block = [rows[i]]
blocks.append(cur_block)

print(f"Total continuous Blocks in v18: {len(blocks)}")

# Calculate Gaps
gaps = []
for i in range(len(blocks) - 1):
    b1 = blocks[i]
    b2 = blocks[i+1]
    p_last = b1[-1]['pid']
    p_next = b2[0]['pid']
    t_last = b1[-1]['ts']
    t_next = b2[0]['ts']
    
    dt1 = get_dt_from_pid(p_last)
    dt2 = get_dt_from_pid(p_next)
    missing = int((dt2 - dt1).total_seconds() / 30.0) - 1
    
    sec = missing * 30
    h_d = sec // 3600
    m_d = (sec % 3600) // 60
    s_d = sec % 60
    dur_str = f"{h_d}:{m_d:02d}:{s_d:02d}"
    
    gaps.append({
        'gap_idx': i + 1,
        'prec_pid': p_last,
        'prec_time': t_last,
        'res_pid': p_next,
        'res_time': t_next,
        'missing_rounds': missing,
        'duration': dur_str,
        'block_before': i + 1,
        'block_after': i + 2
    })

print(f"Total Gaps documented in v18: {len(gaps)}")
print(f"Total missing rounds across all gaps: {sum(g['missing_rounds'] for g in gaps)}")
