def get_api_timestamp(pid: str) -> str:
    year = int(pid[:4])
    month = int(pid[4:6])
    day = int(pid[6:8])
    suffix = int(pid[-5:])
    base = datetime(year, month, day, 5, 30, 0)
    dt = base + timedelta(seconds=(suffix - 50000) * 30)
    return dt.strftime('%Y-%m-%d %H:%M:%S')

# Calculate Blocks and Gaps
def get_day_round(pid):
    suffix = int(pid[-5:]) - 50000
    day = pid[:8]
    return day, suffix

blocks = []
cur_block = [rows[0]]

for i in range(1, len(rows)):
    prev_day, prev_rnd = get_day_round(rows[i-1]['pid'])
    curr_day, curr_rnd = get_day_round(rows[i]['pid'])
    
    is_step = False
    if prev_day == curr_day and curr_rnd == prev_rnd + 1:
        is_step = True
    elif prev_rnd == 2880 and curr_rnd == 1:
        # Seamless transition across midnight UTC
        is_step = True
        
    if is_step:
        cur_block.append(rows[i])
    else:
        blocks.append(cur_block)
        cur_block = [rows[i]]
blocks.append(cur_block)

print(f"Total Blocks in v12: {len(blocks)}")
for b_idx, b in enumerate(blocks, 1):
    print(f"  Block {b_idx:02d}: Seqs {b[0]['seq']}..{b[-1]['seq']} ({len(b)} rounds) | {b[0]['pid']}..{b[-1]['pid']} | {b[0]['ts']} -> {b[-1]['ts']}")

# Calculate Gaps
gaps = []
for i in range(len(blocks) - 1):
    b1 = blocks[i]
    b2 = blocks[i+1]
    p_last = b1[-1]['pid']
    p_next = b2[0]['pid']
    t_last = b1[-1]['ts']
    t_next = b2[0]['ts']
    
    d1, r1 = get_day_round(p_last)
    d2, r2 = get_day_round(p_next)
    
    if d1 == d2:
        missing = r2 - r1 - 1
    else:
        missing = (2880 - r1) + (r2 - 1)
        
    gaps.append({
        'gap_idx': i + 1,
        'prec_pid': p_last,
        'prec_time': t_last,
        'res_pid': p_next,
        'res_time': t_next,
        'missing_rounds': missing,
        'block_before': i + 1,
        'block_after': i + 2
    })

print(f"Total Gaps in v12: {len(gaps)}")
for g in gaps:
    print(f"  Gap {g['gap_idx']:02d}: {g['prec_pid']} -> {g['res_pid']} (missing {g['missing_rounds']} rounds)")
