def get_dt(pid):
    year = int(pid[:4])
    month = int(pid[4:6])
    day = int(pid[6:8])
    suffix = int(pid[-5:])
    base = datetime(year, month, day, 5, 30, 0)
    return base + timedelta(seconds=(suffix - 50000) * 30)

def get_sha256(path):
    h = hashlib.sha256()
    with open(path, 'rb') as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest()
