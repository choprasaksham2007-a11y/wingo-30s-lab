import cv2, os, numpy as np

b2_dir = r"extraction\extracted_zip_batch2"

# Seq 1277 is in WhatsApp Image 2026-10-23 at 22.06.51.jpeg
# Let's inspect Seq 1277 (PID 50455)
img = cv2.imread(os.path.join(b2_dir, "WhatsApp Image 2026-10-23 at 22.06.51.jpeg"))

# Search around y from 500 to 1400
for yc in range(550, 1400, 5):
    crop_wide = img[yc-20:yc+20, 330:410]
    hsv = cv2.cvtColor(crop_wide, cv2.COLOR_BGR2HSV)
    r_mask = cv2.inRange(hsv, np.array([0, 70, 70]), np.array([10, 255, 255])) | cv2.inRange(hsv, np.array([170, 70, 70]), np.array([180, 255, 255]))
    coords = np.argwhere(r_mask > 0)
    if len(coords) > 20:
        min_y, min_x = coords.min(axis=0)
        max_y, max_x = coords.max(axis=0)
        glyph = r_mask[min_y:max_y+1, min_x:max_x+1]
        h, w = glyph.shape
        top_right_box = glyph[int(h*0.15):int(h*0.45), int(w*0.6):]
        tr_fill = cv2.countNonZero(top_right_box)
        if 25 <= tr_fill <= 30 and h > 30:
            print(f"Found sample with tr_fill={tr_fill} at yc={yc}, shape=({h}, {w}):")
            for r in glyph:
                print("".join(["#" if p > 0 else " " for p in r]))
            print("=" * 40)
            break
