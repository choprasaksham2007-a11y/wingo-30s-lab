import pandas as pd
import hashlib

csv_path = r"project_reports\colour_raw_data_v18.csv"
df = pd.read_csv(csv_path, dtype={'Period ID': str})

history = [(str(row['Period ID']).strip(), int(row['number'])) for _, row in df.iterrows()]

def sha_prediction(period):
    digest = hashlib.sha256(period.encode()).digest()
    return int.from_bytes(digest, "big") % 10

split = int(len(history) * 0.8)
train = history[:split]
test = history[split:]

matches = sum(
    sha_prediction(period) == actual
    for period, actual in test
)

accuracy = matches / len(test) if test else 0

from collections import Counter
common_digit = Counter(n for _, n in train).most_common(1)[0][0]
baseline_hits = sum(n == common_digit for _, n in test)

print("=== EXACT USER SCRIPT EXECUTION ON MASTER V18 ===")
print("Training rounds:", len(train))
print("Test rounds:", len(test))
print("SHA-256 matches:", matches)
print(f"SHA-256 accuracy: {accuracy:.2%}")
print(f"Frequency baseline: {baseline_hits / len(test):.2%}")
print("Uniform random baseline: 10.00%")
