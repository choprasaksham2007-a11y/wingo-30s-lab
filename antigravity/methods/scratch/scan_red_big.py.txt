import cv2, os, glob, numpy as np, asyncio, winocr
from PIL import Image

batch4_dir = r"extraction\extracted_zip_batch4"

async def scan_red_big():
    files = sorted(glob.glob(os.path.join(batch4_dir, "*.jpeg")))[:5]
    
    for f in files:
        im = cv2.imread(f)
        pid_strip = im[500:1450, 20:300]
        res_pid = await winocr.recognize_pil(Image.fromarray(cv2.cvtColor(pid_strip, cv2.COLOR_BGR2RGB)))
        
        bs_strip = im[500:1450, 420:530]
        res_bs = await winocr.recognize_pil(Image.fromarray(cv2.cvtColor(bs_strip, cv2.COLOR_BGR2RGB)))
        bs_items = []
        for l in res_bs.lines:
            txt = " ".join([w.text.strip() for w in l.words])
            if "Big" in txt or "Small" in txt:
                val = "Small" if "Small" in txt else "Big"
                y_c = l.words[0].bounding_rect.y + l.words[0].bounding_rect.height / 2.0
                bs_items.append((y_c, val))
                
        for line in res_pid.lines:
            txt = ''.join([w.text.strip() for w in line.words]).replace(' ', '').replace('o', '0').replace('O', '0')
            for i in range(len(txt)-16):
                chunk = txt[i:i+17]
                if chunk.isdigit() and chunk.startswith('202610241000'):
                    w0 = line.words[0]
                    y_rel = w0.bounding_rect.y + w0.bounding_rect.height / 2.0
                    yc = int(round(500 + y_rel))
                    
                    best_bs = None
                    min_d = 999.0
                    for b_y, b_val in bs_items:
                        dist = abs(b_y - y_rel)
                        if dist < min_d and dist < 35:
                            min_d = dist
                            best_bs = b_val
                            
                    crop = im[yc-20:yc+20, 315:375]
                    hsv = cv2.cvtColor(crop, cv2.COLOR_BGR2HSV)
                    g_cnt = cv2.countNonZero(cv2.inRange(hsv, np.array([35, 50, 50]), np.array([85, 255, 255])))
                    r_cnt = cv2.countNonZero(cv2.inRange(hsv, np.array([0, 70, 70]), np.array([10, 255, 255])) | cv2.inRange(hsv, np.array([170, 70, 70]), np.array([180, 255, 255])))
                    v_cnt = cv2.countNonZero(cv2.inRange(hsv, np.array([125, 40, 40]), np.array([165, 255, 255])))
                    
                    if v_cnt < 40 and r_cnt > g_cnt and best_bs == "Big":
                        r_mask = cv2.inRange(hsv, np.array([0, 70, 70]), np.array([10, 255, 255])) | cv2.inRange(hsv, np.array([170, 70, 70]), np.array([180, 255, 255]))
                        coords = np.argwhere(r_mask > 0)
                        if len(coords) > 20:
                            min_y, min_x = coords.min(axis=0)
                            max_y, max_x = coords.max(axis=0)
                            w = max_x - min_x + 1
                            h = max_y - min_y + 1
                            glyph = r_mask[min_y:max_y+1, min_x:max_x+1]
                            # Check top-right of glyph (top 20% to 50%, right 40%)
                            top_right = cv2.countNonZero(glyph[h//5:h//2, int(w*0.6):])
                            print(f"PID {chunk} (Red Big): w={w}, h={h}, top_right={top_right}, file={os.path.basename(f)}")

if __name__ == '__main__':
    asyncio.run(scan_red_big())
