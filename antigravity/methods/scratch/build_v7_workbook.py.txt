# Styling definitions

# ── 1. Raw_Data ──
print("Generating Raw_Data sheet...")
ws_raw = wb.create_sheet(title="Raw_Data")
raw_headers = ["sequence_no", "Period ID", "timestamp", "number", "colour", "size", "colour_original", "fix_note"]
ws_raw.append(raw_headers)
for c_idx in range(1, len(raw_headers) + 1):
    c = ws_raw.cell(row=1, column=c_idx)
    c.fill = FILL_HEADER
    c.font = FONT_HEADER
    c.alignment = Alignment(horizontal="center", vertical="center")

for idx, r in enumerate(rows, start=1):
    row_vals = [
        idx,
        str(r['pid']),
        r['ts'],
        r['num'],
        r['col'],
        r['sz'],
        r['orig_col'],
        r['note']
    ]
    ws_raw.append(row_vals)
    c_pid = ws_raw.cell(row=idx+1, column=2)
    c_pid.number_format = "@"

for col in ws_raw.columns:
    max_len = max(len(str(cell.value or '')) for cell in col)
    col_letter = get_column_letter(col[0].column)
    ws_raw.column_dimensions[col_letter].width = max(max_len + 3, 12)
ws_raw.column_dimensions['B'].width = 22
ws_raw.column_dimensions['C'].width = 20
ws_raw.column_dimensions['H'].width = 45

print("Generating ML_Ready sheet...")
ws_ml = wb.create_sheet(title="ML_Ready")

    c.fill = FILL_HEADER
    c.font = FONT_HEADER
    c.alignment = Alignment(horizontal="center", vertical="center")

sample_idx = 1
ml_rows_list = []
for b_idx, block in enumerate(blocks, 1):
    for i in range(5, len(block)):
        tgt = block[i]
        p1 = block[i-1]
        p2 = block[i-2]
        p3 = block[i-3]
        p4 = block[i-4]
        p5 = block[i-5]
        ml_row = [
            sample_idx,
            str(tgt['pid']),
            tgt['ts'],
            tgt['num'],
            tgt['col'],
            tgt['sz'],
            p1['num'], p1['col'], p1['sz'],
            p2['num'], p2['col'], p2['sz'],
            p3['num'], p3['col'], p3['sz'],
            p4['num'], p4['col'], p4['sz'],
            p5['num'], p5['col'], p5['sz'],
            b_idx
        ]
        ws_ml.append(ml_row)
        c_pid = ws_ml.cell(row=sample_idx+1, column=2)
        c_pid.number_format = "@"
        ml_rows_list.append(ml_row)
        sample_idx += 1

for col in ws_ml.columns:
    max_len = max(len(str(cell.value or '')) for cell in col)
    col_letter = get_column_letter(col[0].column)
    ws_ml.column_dimensions[col_letter].width = max(max_len + 3, 11)
ws_ml.column_dimensions['B'].width = 22

# ── 3. Missing_Rounds ──
print("Generating Missing_Rounds sheet...")
ws_gaps = wb.create_sheet(title="Missing_Rounds")
gap_headers = [
    "Gap #", "Preceding Period ID", "Preceding Time", "Resuming Period ID", "Resuming Time",
    "Missing Rounds", "Duration", "Block Before", "Block After"
]
ws_gaps.append(gap_headers)
for c_idx in range(1, len(gap_headers) + 1):
    c = ws_gaps.cell(row=1, column=c_idx)
    c.fill = FILL_HEADER
    c.font = FONT_HEADER
    c.alignment = Alignment(horizontal="center", vertical="center")

for g in gaps:
    sec = g['missing_rounds'] * 30
    h_d = sec // 3600
    m_d = (sec % 3600) // 60
    s_d = sec % 60
    dur_str = f"{h_d}:{m_d:02d}:{s_d:02d}"
    ws_gaps.append([
        g['gap_idx'],
        str(g['prec_pid']),
        g['prec_time'],
        str(g['res_pid']),
        g['res_time'],
        g['missing_rounds'],
        dur_str,
        g['block_before'],
        g['block_after']
    ])

for col in ws_gaps.columns:
    max_len = max(len(str(cell.value or '')) for cell in col)
    col_letter = get_column_letter(col[0].column)
    ws_gaps.column_dimensions[col_letter].width = max(max_len + 3, 14)
ws_gaps.column_dimensions['B'].width = 22
ws_gaps.column_dimensions['D'].width = 22
