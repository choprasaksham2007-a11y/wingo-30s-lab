print("--> WAY 10: Multi-Directory SHA-256 Hash Invariance Check...")
def get_sha256(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while chunk := f.read(65536):
            h.update(chunk)
    return h.hexdigest()
