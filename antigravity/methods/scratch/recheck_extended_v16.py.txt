import openpyxl, os, sys, csv
import pandas as pd
from datetime import datetime

sys.stdout.reconfigure(encoding='utf-8')

brain_dir = r"project_reports"
excel_path = os.path.join(brain_dir, "Colour_Prediction_Dataset_v16.xlsx")
raw_csv_path = os.path.join(brain_dir, "colour_raw_data_v16.csv")
ml_csv_path = os.path.join(brain_dir, "colour_ml_ready_v16.csv")

print("=" * 80)
print("  EXTENDED PASS-2 INDEPENDENT RECHECK AUDIT - MASTER DATASET V16")
print("=" * 80)

wb = openpyxl.load_workbook(excel_path, data_only=True)

# 1. Sheet presence and structure
expected_sheets = [
    "Raw_Data", "ML_Ready", "Missing_Rounds", "Summary", "Read_Me",
    "Audit", "ID_Audit", "User_Provided", "Change_Log", "Audit_8_Rounds"
]
actual_sheets = wb.sheetnames
assert actual_sheets == expected_sheets, f"Sheet mismatch: {actual_sheets} vs {expected_sheets}"
print(f"[PASS] Sheet Structure: All 10 expected sheets present in exact sequence.")

# 2. Raw_Data vs colour_raw_data_v16.csv line-by-line byte/value comparison
ws_raw = wb['Raw_Data']
excel_raw_rows = list(ws_raw.iter_rows(values_only=True))
with open(raw_csv_path, 'r', encoding='utf-8') as f:
    csv_raw_rows = list(csv.reader(f))

assert len(excel_raw_rows) == len(csv_raw_rows), f"Row count mismatch: {len(excel_raw_rows)} vs {len(csv_raw_rows)}"
assert excel_raw_rows[0] == tuple(csv_raw_rows[0]), f"Header mismatch: {excel_raw_rows[0]} vs {csv_raw_rows[0]}"

for i in range(1, len(excel_raw_rows)):
    er = [str(x) if x is not None else '' for x in excel_raw_rows[i]]
    cr = csv_raw_rows[i]
    if er != cr:
        raise AssertionError(f"Mismatch at raw row {i}: Excel={er} vs CSV={cr}")

print(f"[PASS] Raw_Data Parity: 100% exact match across all {len(csv_raw_rows)-1} data rows between Excel and CSV.")
