import pandas as pd
import numpy as np
from scipy import stats
import json
import sys

excel_path = r"project_reports\Colour_Prediction_Dataset_v9.xlsx"
df = pd.read_excel(excel_path, sheet_name='Raw_Data')

N = len(df)
print(f"Total Rows: {N}")

# 1. Digit frequency distribution & Chi-Square Goodness-of-Fit
digit_counts = df['number'].value_counts().sort_index()
expected_count = N / 10.0

digit_dist_dict = {str(d): int(digit_counts.get(d, 0)) for d in range(10)}
digit_pct_dict = {str(d): round(digit_counts.get(d, 0) / N * 100.0, 2) for d in range(10)}

chi2_stat, p_val = stats.chisquare(f_obs=digit_counts.values, f_exp=[expected_count]*10)
df_chi2 = 10 - 1

print("\n--- Digit Frequency Distribution ---")
for d in range(10):
    c = digit_dist_dict[str(d)]
    pct = digit_pct_dict[str(d)]
    diff = c - expected_count
    print(f"Digit {d}: Count={c:4d} ({pct:5.2f}%), Exp={expected_count:.1f}, Diff={diff:+5.1f}")

print(f"\nChi-Square Test (Uniformity): chi2={chi2_stat:.4f}, df={df_chi2}, p_value={p_val:.4f}")

# 2. Big vs Small distribution & Parity
size_counts = df['size'].value_counts()
big_count = int(size_counts.get('Big', 0))
small_count = int(size_counts.get('Small', 0))
big_pct = round(big_count / N * 100.0, 2)
small_pct = round(small_count / N * 100.0, 2)

chi2_size, p_val_size = stats.chisquare(f_obs=[big_count, small_count], f_exp=[N/2.0, N/2.0])

print(f"\n--- Big vs Small Distribution ---")
print(f"Big:   {big_count} ({big_pct}%)")
print(f"Small: {small_count} ({small_pct}%)")
print(f"Size Chi-Square: chi2={chi2_size:.4f}, p_value={p_val_size:.4f}")

# Colour distribution
colour_counts = df['colour'].value_counts()
colour_dist_dict = {str(k): int(v) for k, v in colour_counts.items()}
print(f"\n--- Colour Distribution ---")
for col, cnt in colour_dist_dict.items():
    print(f"  {col}: {cnt} ({cnt/N*100.0:.2f}%)")

# 3. First-order transition matrix (excluding gaps > 30s)
# First let's detect gaps in sequence / period ID
df['dt'] = pd.to_datetime(df['timestamp'])
df['dt_diff'] = df['dt'].diff()
df['pid_diff'] = df['Period ID'].diff()

transitions = {'Big': {'Big': 0, 'Small': 0}, 'Small': {'Big': 0, 'Small': 0}}
total_trans = 0

for i in range(len(df) - 1):
    # Only count continuous transitions (interval == 30s and pid_diff == 1)
    # Check if next round is continuous
    if df.iloc[i+1]['dt_diff'] == pd.Timedelta(seconds=30) and df.iloc[i+1]['pid_diff'] == 1:
        curr_sz = df.iloc[i]['size']
        next_sz = df.iloc[i+1]['size']
        transitions[curr_sz][next_sz] += 1
        total_trans += 1

bb = transitions['Big']['Big']
bs = transitions['Big']['Small']
sb = transitions['Small']['Big']
ss = transitions['Small']['Small']

p_bb = round(bb / (bb + bs), 4) if (bb + bs) > 0 else 0
p_bs = round(bs / (bb + bs), 4) if (bb + bs) > 0 else 0
p_sb = round(sb / (sb + ss), 4) if (sb + ss) > 0 else 0
p_ss = round(ss / (sb + ss), 4) if (sb + ss) > 0 else 0

print(f"\n--- First-Order Transition Matrix (Continuous Transitions, N={total_trans}) ---")
print(f"P(Big -> Big):     {p_bb:.4f} (count={bb})")
print(f"P(Big -> Small):   {p_bs:.4f} (count={bs})")
print(f"P(Small -> Big):   {p_sb:.4f} (count={sb})")
print(f"P(Small -> Small): {p_ss:.4f} (count={ss})")

# Also global transitions (all consecutive rows)
trans_global = {'Big': {'Big': 0, 'Small': 0}, 'Small': {'Big': 0, 'Small': 0}}
for i in range(len(df) - 1):
    curr_sz = df.iloc[i]['size']
    next_sz = df.iloc[i+1]['size']
    trans_global[curr_sz][next_sz] += 1

print(f"\n--- Global Row Transitions (N={len(df)-1}) ---")
bb_g = trans_global['Big']['Big']
bs_g = trans_global['Big']['Small']
sb_g = trans_global['Small']['Big']
ss_g = trans_global['Small']['Small']
print(f"P(Big -> Big):     {bb_g/(bb_g+bs_g):.4f} (count={bb_g})")
print(f"P(Big -> Small):   {bs_g/(bb_g+bs_g):.4f} (count={bs_g})")
print(f"P(Small -> Big):   {sb_g/(sb_g+ss_g):.4f} (count={sb_g})")
print(f"P(Small -> Small): {ss_g/(sb_g+ss_g):.4f} (count={ss_g})")

# 4. Streak Analysis
# Calculate longest streaks for:
# - Big
# - Small
# - Pure Red (2, 4, 6, 8)
# - Pure Green (1, 3, 7, 9)
# - Red Family (Red, Red+Violet)
# - Green Family (Green, Green+Violet)
# - Number Repeats (identical consecutive digits)

def find_streaks(condition_series):
    streaks = []
    curr_len = 0
    start_idx = 0
    for idx, val in enumerate(condition_series):
        if val:
            if curr_len == 0:
                start_idx = idx
            curr_len += 1
        else:
            if curr_len > 0:
                streaks.append((curr_len, start_idx, idx - 1))
                curr_len = 0
    if curr_len > 0:
        streaks.append((curr_len, start_idx, len(condition_series) - 1))
    return sorted(streaks, key=lambda x: x[0], reverse=True)

big_streaks = find_streaks(df['size'] == 'Big')
small_streaks = find_streaks(df['size'] == 'Small')
pure_red_streaks = find_streaks(df['colour'] == 'Red')
pure_green_streaks = find_streaks(df['colour'] == 'Green')
red_fam_streaks = find_streaks(df['colour'].isin(['Red', 'Red+Violet']))
green_fam_streaks = find_streaks(df['colour'].isin(['Green', 'Green+Violet']))

# Number repeats
num_repeats = []
curr_num = df.iloc[0]['number']
curr_len = 1
start_idx = 0
for idx in range(1, len(df)):
    num = df.iloc[idx]['number']
    if num == curr_num:
        curr_len += 1
    else:
        if curr_len >= 2:
            num_repeats.append((curr_len, curr_num, start_idx, idx - 1))
        curr_num = num
        curr_len = 1
        start_idx = idx
if curr_len >= 2:
    num_repeats.append((curr_len, curr_num, start_idx, len(df) - 1))

num_repeats.sort(key=lambda x: x[0], reverse=True)

# Count repeats breakdown
repeat_counts = {2: 0, 3: 0, 4: 0, '5+': 0}
for length, digit, s_idx, e_idx in num_repeats:
    if length == 2:
        repeat_counts[2] += 1
    elif length == 3:
        repeat_counts[3] += 1
    elif length == 4:
        repeat_counts[4] += 1
    else:
        repeat_counts['5+'] += 1

print("\n--- Streak Analysis ---")
print(f"Longest Big streak:   {big_streaks[0][0]} rounds (Seq {df.iloc[big_streaks[0][1]]['sequence_no']} to {df.iloc[big_streaks[0][2]]['sequence_no']}, PID {df.iloc[big_streaks[0][1]]['Period ID']} to {df.iloc[big_streaks[0][2]]['Period ID']})")
print(f"Longest Small streak: {small_streaks[0][0]} rounds (Seq {df.iloc[small_streaks[0][1]]['sequence_no']} to {df.iloc[small_streaks[0][2]]['sequence_no']}, PID {df.iloc[small_streaks[0][1]]['Period ID']} to {df.iloc[small_streaks[0][2]]['Period ID']})")
print(f"Longest Pure Red streak:   {pure_red_streaks[0][0]} rounds (Seq {df.iloc[pure_red_streaks[0][1]]['sequence_no']} to {df.iloc[pure_red_streaks[0][2]]['sequence_no']})")
print(f"Longest Pure Green streak: {pure_green_streaks[0][0]} rounds (Seq {df.iloc[pure_green_streaks[0][1]]['sequence_no']} to {df.iloc[pure_green_streaks[0][2]]['sequence_no']})")
print(f"Longest Red-Family streak:   {red_fam_streaks[0][0]} rounds (Seq {df.iloc[red_fam_streaks[0][1]]['sequence_no']} to {df.iloc[red_fam_streaks[0][2]]['sequence_no']})")
print(f"Longest Green-Family streak: {green_fam_streaks[0][0]} rounds (Seq {df.iloc[green_fam_streaks[0][1]]['sequence_no']} to {df.iloc[green_fam_streaks[0][2]]['sequence_no']})")
top_repeat = num_repeats[0]
print(f"Longest Number Repeat: {top_repeat[0]} rounds of digit {top_repeat[1]} (Seq {df.iloc[top_repeat[2]]['sequence_no']} to {df.iloc[top_repeat[3]]['sequence_no']}, PID {df.iloc[top_repeat[2]]['Period ID']} to {df.iloc[top_repeat[3]]['Period ID']})")
print(f"Number Repeat Breakdown: 2x={repeat_counts[2]}, 3x={repeat_counts[3]}, 4x={repeat_counts[4]}, 5x+={repeat_counts['5+']}")

stat_results = {
    'total_rows': N,
    'digit_distribution': digit_dist_dict,
    'digit_percentages': digit_pct_dict,
    'chi_square_uniform': {
        'statistic': round(float(chi2_stat), 4),
        'degrees_of_freedom': int(df_chi2),
        'p_value': round(float(p_val), 4),
        'is_uniform': bool(p_val > 0.05)
    },
    'size_distribution': {
        'big': big_count,
        'small': small_count,
        'big_pct': big_pct,
        'small_pct': small_pct,
        'chi_square': round(float(chi2_size), 4),
        'p_value': round(float(p_val_size), 4)
    },
    'colour_distribution': colour_dist_dict,
    'first_order_transition_matrix': {
        'P_Big_to_Big': p_bb,
        'P_Big_to_Small': p_bs,
        'P_Small_to_Big': p_sb,
        'P_Small_to_Small': p_ss,
        'counts': {
            'Big_to_Big': bb,
            'Big_to_Small': bs,
            'Small_to_Big': sb,
            'Small_to_Small': ss
        }
    },
    'streak_analysis': {
        'longest_big_streak': int(big_streaks[0][0]),
        'longest_small_streak': int(small_streaks[0][0]),
        'longest_pure_red_streak': int(pure_red_streaks[0][0]),
        'longest_pure_green_streak': int(pure_green_streaks[0][0]),
        'longest_red_family_streak': int(red_fam_streaks[0][0]),
        'longest_green_family_streak': int(green_fam_streaks[0][0]),
        'longest_number_repeat': {
            'digit': int(top_repeat[1]),
            'length': int(top_repeat[0]),
            'start_seq': int(df.iloc[top_repeat[2]]['sequence_no']),
            'end_seq': int(df.iloc[top_repeat[3]]['sequence_no'])
        },
        'number_repeat_breakdown': repeat_counts
    }
}

out_stat = r"scratch\statistical_tests_v9.json"
with open(out_stat, "w") as f:
    json.dump(stat_results, f, indent=2)
print(f"\nSaved statistical tests to {out_stat}")
