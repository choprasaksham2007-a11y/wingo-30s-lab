import pandas as pd
from datetime import datetime, timedelta

csv_path = r"colour-ai\wingo_investigation\collected_data\wingo_30s_history.csv"
df = pd.read_csv(csv_path, dtype={"issueNumber": str, "premium": str, "number": str, "colour": str, "size": str})
print(f"Loaded {len(df)} rows from CSV.")

def get_dt_from_pid(pid: str) -> datetime:
    year = int(pid[:4])
    month = int(pid[4:6])
    day = int(pid[6:8])
    daily_round = int(pid[13:])
    # 05:30:00 IST base for daily round 1
    base = datetime(year, month, day, 5, 30, 0)
    return base + timedelta(seconds=(daily_round - 1) * 30)

pids = df["issueNumber"].tolist()
blocks = []
cur_block = [pids[0]]

for i in range(1, len(pids)):
    dt_prev = get_dt_from_pid(pids[i-1])
    dt_curr = get_dt_from_pid(pids[i])
    if (dt_curr - dt_prev).total_seconds() == 30.0:
        cur_block.append(pids[i])
    else:
        blocks.append(cur_block)
        cur_block = [pids[i]]
blocks.append(cur_block)

print(f"Total Continuous Blocks: {len(blocks)}")
print(f"Total Gaps between blocks: {len(blocks) - 1}")
block_lens = [len(b) for b in blocks]
print(f"Block lengths: min={min(block_lens)}, max={max(block_lens)}, avg={sum(block_lens)/len(block_lens):.1f}")

# Check missing rounds
total_missing = 0
for i in range(len(blocks) - 1):
    dt1 = get_dt_from_pid(blocks[i][-1])
    dt2 = get_dt_from_pid(blocks[i+1][0])
    miss = int((dt2 - dt1).total_seconds() / 30.0) - 1
    total_missing += miss

print(f"Total missing rounds across all gaps: {total_missing}")

# ML training samples with lag-5
ml_samples = sum(max(0, len(b) - 5) for b in blocks)
print(f"ML Ready samples (lag-5, 0 gap crossings): {ml_samples}")
