import zipfile
import os

zip_path = r'project_reports\inputs\input_bce4b90b8c.zip'
dest_dir = r'extraction\extracted_zip_batch7'

os.makedirs(dest_dir, exist_ok=True)
with zipfile.ZipFile(zip_path, 'r') as zf:
    zf.extractall(dest_dir)

files = [f for f in os.listdir(dest_dir) if os.path.isfile(os.path.join(dest_dir, f))]
print(f"Extracted {len(files)} files to {dest_dir}")
for f in sorted(files)[:10]:
    print("Sample file:", f)
