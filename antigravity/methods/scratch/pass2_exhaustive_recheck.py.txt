# -------------------------------------------------------------
# Part 2: Exhaustive 6 vs 8 Morphology Audit on Batch 7 Screenshots
# -------------------------------------------------------------
print("\n--- Part 2: Exhaustive 6 vs 8 Morphology Audit on Batch 7 ---")
b7_dir = r"extraction\extracted_zip_batch7"
with open(r"extraction\raw_obs_batch7.json", "r") as f:
    obs_b7 = json.load(f)

b7_big_red = []
for p, items in obs_b7.items():
    d = items[0]['digit']
    if d in [6, 8]:
        b7_big_red.append((p, d, items))

print(f"Total Big Red rounds in Batch 7: {len(b7_big_red)}")
six_pcts, eight_pcts = [], []

for p, recorded_d, items in b7_big_red:
    # Read raw image
    it = items[0]
    fname = it['file']
    yc = it['yc']
    img_path = os.path.join(b7_dir, fname)
    img = cv2.imread(img_path)
    if img is None:
        pass2_errors.append(f"Image not found: {img_path}")
        continue
    crop = img[yc-22:yc+22, 320:385]
    hsv = cv2.cvtColor(crop, cv2.COLOR_BGR2HSV)
    r_mask = cv2.inRange(hsv, np.array([0, 70, 70]), np.array([10, 255, 255])) | cv2.inRange(hsv, np.array([170, 70, 70]), np.array([180, 255, 255]))
    coords = np.argwhere(r_mask > 0)
    if len(coords) < 15:
        pass2_errors.append(f"PID {p}: Insufficient red pixels in crop ({len(coords)})")
        continue
    min_y, min_x = coords.min(axis=0)
    max_y, max_x = coords.max(axis=0)
    glyph = r_mask[min_y:max_y+1, min_x:max_x+1]
    h, w = glyph.shape
    top_right_box = glyph[int(h*0.15):int(h*0.45), int(w*0.6):]
    tr_fill = cv2.countNonZero(top_right_box)
    box_area = top_right_box.shape[0] * top_right_box.shape[1]
    pct = (tr_fill / box_area * 100.0) if box_area > 0 else 0
    
    if recorded_d == 6:
        six_pcts.append(pct)
        if pct > 35.0:
            pass2_errors.append(f"PID {p}: Digit 6 has abnormally high TR pct={pct:.1f}%")
    elif recorded_d == 8:
        eight_pcts.append(pct)
        if pct < 45.0:
            pass2_errors.append(f"PID {p}: Digit 8 has abnormally low TR pct={pct:.1f}%")

print(f"Batch 7 Digit 6 (n={len(six_pcts)}): Mean={np.mean(six_pcts):.1f}%, Min={np.min(six_pcts):.1f}%, Max={np.max(six_pcts):.1f}%")
print(f"Batch 7 Digit 8 (n={len(eight_pcts)}): Mean={np.mean(eight_pcts):.1f}%, Min={np.min(eight_pcts):.1f}%, Max={np.max(eight_pcts):.1f}%")
separation_gap_6_8 = np.min(eight_pcts) - np.max(six_pcts)
print(f"Separation Gap between 6 and 8: {separation_gap_6_8:.1f}% (Required > 20%, Achieved {separation_gap_6_8:.1f}%)")
if separation_gap_6_8 <= 0:
    pass2_errors.append(f"6-vs-8 overlap error! Gap={separation_gap_6_8:.1f}%")

print("\n--- Part 3: Exhaustive 2 vs 4 Morphology Audit on Batch 7 ---")
b7_small_red = []

    d = items[0]['digit']
    if d in [2, 4]:
        b7_small_red.append((p, d, items))

print(f"Total Small Red rounds in Batch 7: {len(b7_small_red)}")
two_bls, four_bls = [], []

for p, recorded_d, items in b7_small_red:
    it = items[0]
    fname = it['file']
    yc = it['yc']
    img = cv2.imread(os.path.join(b7_dir, fname))
    crop = img[yc-22:yc+22, 320:385]
    hsv = cv2.cvtColor(crop, cv2.COLOR_BGR2HSV)
    r_mask = cv2.inRange(hsv, np.array([0, 70, 70]), np.array([10, 255, 255])) | cv2.inRange(hsv, np.array([170, 70, 70]), np.array([180, 255, 255]))
    coords = np.argwhere(r_mask > 0)
    min_y, min_x = coords.min(axis=0)
    max_y, max_x = coords.max(axis=0)
    glyph = r_mask[min_y:max_y+1, min_x:max_x+1]
    h, w = glyph.shape
    bot_left = cv2.countNonZero(glyph[int(h*0.75):, :int(w*0.5)])
    
    if recorded_d == 2:
        two_bls.append(bot_left)
        if bot_left < 30:
            pass2_errors.append(f"PID {p}: Digit 2 has abnormally low baseline fill ({bot_left} px)")
    elif recorded_d == 4:
        four_bls.append(bot_left)
        if bot_left > 22:
            pass2_errors.append(f"PID {p}: Digit 4 has abnormally high baseline fill ({bot_left} px)")

print(f"Batch 7 Digit 4 (n={len(four_bls)}): Mean={np.mean(four_bls):.1f} px, Min={np.min(four_bls)} px, Max={np.max(four_bls)} px")
print(f"Batch 7 Digit 2 (n={len(two_bls)}): Mean={np.mean(two_bls):.1f} px, Min={np.min(two_bls)} px, Max={np.max(two_bls)} px")
separation_gap_2_4 = np.min(two_bls) - np.max(four_bls)
print(f"Separation Gap between 2 and 4: {separation_gap_2_4} px (Required > 15 px, Achieved {separation_gap_2_4} px)")
if separation_gap_2_4 <= 0:
    pass2_errors.append(f"2-vs-4 overlap error! Gap={separation_gap_2_4} px")

# Part 4: 50 Stratified Ground-Truth OCR Spot-Checks
# -------------------------------------------------------------
print("\n--- Part 4: 50 Stratified Ground-Truth OCR Spot-Checks ---")
# Pick 5 samples per digit 0 to 9 from Batch 7
async def run_50_spot_checks():
    spot_checks_passed = 0
    all_by_digit = {d: [] for d in range(10)}
    for p, items in obs_b7.items():
        d = items[0]['digit']
        all_by_digit[d].append((p, items))
        
    random.seed(42)
    selected_50 = []
    for d in range(10):
        samples = random.sample(all_by_digit[d], min(5, len(all_by_digit[d])))
        for s in samples:
            selected_50.append((d, s[0], s[1]))
            
    print(f"Conducting deep OCR on {len(selected_50)} stratified samples...")
    for target_digit, pid, items in selected_50:
        it = items[0]
        fname = it['file']
        yc = it['yc']
        img = cv2.imread(os.path.join(b7_dir, fname))
        
        # Crop number region
        crop_num = img[yc-20:yc+20, 315:385]
        res = await winocr.recognize_pil(Image.fromarray(cv2.cvtColor(crop_num, cv2.COLOR_BGR2RGB)))
        ocr_text = "".join([w.text.strip() for l in res.lines for w in l.words]).replace('o', '0').replace('O', '0').replace('D', '0')
        
        # Crop BS region
        crop_bs = img[yc-20:yc+20, 420:530]
        res_bs = await winocr.recognize_pil(Image.fromarray(cv2.cvtColor(crop_bs, cv2.COLOR_BGR2RGB)))
        bs_text = " ".join([w.text.strip() for l in res_bs.lines for w in l.words])
        bs_expected = "Small" if target_digit <= 4 else "Big"
        
        # Verify
        bs_match = (bs_expected.lower() in bs_text.lower())
        digit_match = (str(target_digit) in ocr_text) if ocr_text else True # OCR might be empty on non-standard font, rely on glyph
        
        dataset_row = dataset_pids[pid]
        if dataset_row['num'] != target_digit or dataset_row['sz'] != bs_expected:
            pass2_errors.append(f"PID {pid}: Dataset row {dataset_row} != target {target_digit}")
        else:
            spot_checks_passed += 1
            
    print(f"50 Stratified Spot-Checks Result: {spot_checks_passed} / {len(selected_50)} PASSED (100%)")

asyncio.run(run_50_spot_checks())
