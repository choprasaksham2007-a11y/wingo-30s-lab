sub_v7 = df_v7[(df_v7['Period ID'] >= overlap_start_pid) & (df_v7['Period ID'] <= overlap_end_pid)].copy()
sub_v8 = df_v8[(df_v8['Period ID'] >= overlap_start_pid) & (df_v8['Period ID'] <= overlap_end_pid)].copy()

sub_v8 = sub_v8.sort_values('Period ID')

# 3. Morphology Deep Audit on Batch 7 Candidates:
with open(raw_b7_path, 'r') as f:
    obs = json.load(f)

print("\n--- AUDITING MORPHOLOGY METRICS IN BATCH 7 ---")

six_tr_pcts = []
eight_tr_pcts = []
two_bl_fills = []
four_bl_fills = []

for pid_str, items in obs.items():
    pid = int(pid_str)
    is_new = (pid >= 20261024100051294)
    it = items[0]
    assigned_digit = it['digit']
    fname = it['file']
    yc = it['yc']
    
    im_path = os.path.join(b7_dir, fname)
    if not os.path.exists(im_path):
        continue
    im = cv2.imread(im_path)
    crop = im[yc-22:yc+22, 320:385]
    hsv = cv2.cvtColor(crop, cv2.COLOR_BGR2HSV)
    
    r_mask = cv2.inRange(hsv, np.array([0, 70, 70]), np.array([10, 255, 255])) | cv2.inRange(hsv, np.array([170, 70, 70]), np.array([180, 255, 255]))
    coords_r = np.argwhere(r_mask > 0)
    
    if assigned_digit in [6, 8] and len(coords_r) > 15:
        min_y, min_x = coords_r.min(axis=0)
        max_y, max_x = coords_r.max(axis=0)
        glyph = r_mask[min_y:max_y+1, min_x:max_x+1]
        h, w = glyph.shape
        top_right_box = glyph[int(h*0.15):int(h*0.45), int(w*0.6):]
        tr_fill = cv2.countNonZero(top_right_box)
        box_area = top_right_box.shape[0] * top_right_box.shape[1]
        pct = (tr_fill / box_area * 100.0) if box_area > 0 else 0
        
        contours, _ = cv2.findContours(glyph.copy(), cv2.RETR_CCOMP, cv2.CHAIN_APPROX_SIMPLE)
        num_holes = len(contours) - 1 if contours else 0
        
        if assigned_digit == 6:
            six_tr_pcts.append({'pid': pid, 'is_new': is_new, 'pct': pct, 'file': fname, 'holes': num_holes})
        else:
            eight_tr_pcts.append({'pid': pid, 'is_new': is_new, 'pct': pct, 'file': fname, 'holes': num_holes})
            
    elif assigned_digit in [2, 4] and len(coords_r) > 15:
        min_y, min_x = coords_r.min(axis=0)
        max_y, max_x = coords_r.max(axis=0)
        glyph = r_mask[min_y:max_y+1, min_x:max_x+1]
        h, w = glyph.shape
        bot_left = cv2.countNonZero(glyph[int(h*0.75):, :int(w*0.5)])
        
        if assigned_digit == 2:
            two_bl_fills.append({'pid': pid, 'is_new': is_new, 'bl': bot_left, 'file': fname, 'w': w, 'h': h})
        else:
            four_bl_fills.append({'pid': pid, 'is_new': is_new, 'bl': bot_left, 'file': fname, 'w': w, 'h': h})

print(f"\n--- ENTIRE BATCH 7: 6 vs 8 AUDIT (Total 6s: {len(six_tr_pcts)}, Total 8s: {len(eight_tr_pcts)}) ---")
six_pcts = [x['pct'] for x in six_tr_pcts]
eight_pcts = [x['pct'] for x in eight_tr_pcts]

print(f"Digit 6 TR Density: Min = {np.min(six_pcts):.1f}%, Max = {np.max(six_pcts):.1f}%, Mean = {np.mean(six_pcts):.1f}%, Std = {np.std(six_pcts):.1f}%")
print(f"Digit 8 TR Density: Min = {np.min(eight_pcts):.1f}%, Max = {np.max(eight_pcts):.1f}%, Mean = {np.mean(eight_pcts):.1f}%, Std = {np.std(eight_pcts):.1f}%")

sep_6_8 = np.min(eight_pcts) - np.max(six_pcts)
print(f"Separation Gap (Min 8 - Max 6): {sep_6_8:.1f}%")
assert sep_6_8 > 30.0, f"Separation gap too narrow: {sep_6_8}"
print(">> CERTIFIED: 6 vs 8 separation gap is 100% distinct with zero overlap across all Batch 7 rounds!")

print(f"\n--- ENTIRE BATCH 7: 2 vs 4 AUDIT (Total 2s: {len(two_bl_fills)}, Total 4s: {len(four_bl_fills)}) ---")
two_bls = [x['bl'] for x in two_bl_fills]
four_bls = [x['bl'] for x in four_bl_fills]

print(f"Digit 2 BL Fill: Min = {np.min(two_bls)}, Max = {np.max(two_bls)}, Mean = {np.mean(two_bls):.1f}, Std = {np.std(two_bls):.1f}")
print(f"Digit 4 BL Fill: Min = {np.min(four_bls)}, Max = {np.max(four_bls)}, Mean = {np.mean(four_bls):.1f}, Std = {np.std(four_bls):.1f}")

sep_2_4 = np.min(two_bls) - np.max(four_bls)
print(f"Separation Gap (Min 2 - Max 4): {sep_2_4} px")
assert sep_2_4 > 25, f"Separation gap too narrow: {sep_2_4}"
print(">> CERTIFIED: 2 vs 4 separation gap is 100% distinct with zero overlap across all Batch 7 rounds!")
