import openpyxl
import pandas as pd

def check_block_boundaries(filepath, label):
    print("=" * 80)
    print(f"BLOCK BOUNDARIES AUDIT: {label}")
    print("=" * 80)
    wb = openpyxl.load_workbook(filepath, data_only=True)
    
    # Raw_Data
    ws_raw = wb['Raw_Data']
    raw_headers = [cell.value for cell in ws_raw[1]]
    df_raw = pd.DataFrame([{raw_headers[c-1]: ws_raw.cell(r, c).value for c in range(1, len(raw_headers)+1)} for r in range(2, ws_raw.max_row+1)])
    
    # Missing_Rounds
    ws_miss = wb['Missing_Rounds']
    m_headers = [cell.value for cell in ws_miss[1]]
    gaps = []
    for r in range(2, ws_miss.max_row+1):
        row_vals = [ws_miss.cell(r, c).value for c in range(1, ws_miss.max_column+1)]
        if not any(v is not None for v in row_vals) or any(str(v).startswith("Total") for v in row_vals if v is not None):
            continue
        row_dict = {m_headers[c-1]: ws_miss.cell(r, c).value for c in range(1, len(m_headers)+1)}
        b_id = str(row_dict.get('Before Period ID') or row_dict.get('prior_period_id'))
        a_id = str(row_dict.get('After Period ID') or row_dict.get('subsequent_period_id'))
        gaps.append((b_id, a_id))
        
    # ML_Ready
    ws_ml = wb['ML_Ready']
    ml_headers = [cell.value for cell in ws_ml[1]]
    df_ml = pd.DataFrame([{ml_headers[c-1]: ws_ml.cell(r, c).value for c in range(1, len(ml_headers)+1)} for r in range(2, ws_ml.max_row+1)])
    
    # Find raw indices of gap boundaries
    raw_id_to_idx = {str(df_raw.iloc[i]['Period ID']): i for i in range(len(df_raw))}
    ml_targets = set(df_ml['Target Period ID'].astype(str))
    
    for gap_num, (b_id, a_id) in enumerate(gaps, 1):
        b_idx = raw_id_to_idx[b_id]
        a_idx = raw_id_to_idx[a_id]
        print(f"\nGap #{gap_num}: {b_id} (raw idx {b_idx}) -> {a_id} (raw idx {a_idx})")
        for offset in range(0, 5):
            candidate_id = str(df_raw.iloc[a_idx + offset]['Period ID'])
            is_in_ml = candidate_id in ml_targets
            print(f"  Offset +{offset} ({candidate_id}): In ML_Ready? {is_in_ml} (Must be False!)")
            assert not is_in_ml, f"Violation! Offset {offset} found in ML_Ready!"
            
        first_valid_target_id = str(df_raw.iloc[a_idx + 5]['Period ID'])
        is_first_valid_in_ml = first_valid_target_id in ml_targets
        print(f"  Offset +5 (First Valid: {first_valid_target_id}): In ML_Ready? {is_first_valid_in_ml} (Must be True!)")
        assert is_first_valid_in_ml, f"Violation! First valid target not found in ML_Ready!"
        
    print("\nALL GAP BOUNDARIES CONFIRMED PERFECTLY ENFORCED!")
    wb.close()

check_block_boundaries(r'project_reports\Colour_Prediction_Dataset_corrected.xlsx', 'Colour_Prediction_Dataset_corrected.xlsx')
