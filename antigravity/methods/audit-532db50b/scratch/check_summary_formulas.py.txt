import openpyxl

for fname in ['Colour_Prediction_Dataset_v3.xlsx', 'Colour_Prediction_Dataset_corrected.xlsx']:
    path = rf"project_reports\{fname}"
    wb = openpyxl.load_workbook(path, data_only=False)
    ws = wb['Summary']
    print(f"=== {fname} Summary formulas ===")
    for r in range(1, ws.max_row+1):
        for c in range(1, ws.max_column+1):
            val = str(ws.cell(r, c).value)
            if val.startswith('='):
                print(f"  Cell ({r},{c}) {ws.cell(r,c).coordinate}: {val}")
    wb.close()
