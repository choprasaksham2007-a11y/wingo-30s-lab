import openpyxl
import pandas as pd

def check_rules(val_num, val_col, val_siz):
    # Rule mapping
    if val_num == 0:
        exp_col, exp_siz = 'Red+Violet', 'Small'
    elif val_num == 5:
        exp_col, exp_siz = 'Green+Violet', 'Big'
    elif val_num in [1, 3, 7, 9]:
        exp_col = 'Green'
        exp_siz = 'Small' if val_num < 5 else 'Big'
    elif val_num in [2, 4, 6, 8]:
        exp_col = 'Red'
        exp_siz = 'Small' if val_num < 5 else 'Big'
    else:
        return False, f"Invalid number {val_num}"
        
    if val_col != exp_col:
        return False, f"Colour mismatch for {val_num}: got {val_col}, expected {exp_col}"
    if val_siz != exp_siz:
        return False, f"Size mismatch for {val_num}: got {val_siz}, expected {exp_siz}"
    return True, "OK"

for fname in ['Colour_Prediction_Dataset_v3.xlsx', 'Colour_Prediction_Dataset_corrected.xlsx']:
    path = rf"project_reports\{fname}"
    wb = openpyxl.load_workbook(path, data_only=True)
    ws_ml = wb['ML_Ready']
    headers = [cell.value for cell in ws_ml[1]]
    
    rule_violations = []
    for r in range(2, ws_ml.max_row+1):
        row = {headers[c-1]: ws_ml.cell(r, c).value for c in range(1, len(headers)+1)}
        # check target
        ok, msg = check_rules(row['target_number'], row['target_colour'], row['target_size'])
        if not ok:
            rule_violations.append((r, 'target', msg))
        # check lags 1..5
        for lag in range(1, 6):
            ok, msg = check_rules(row[f'prev_{lag}_number'], row[f'prev_{lag}_colour'], row[f'prev_{lag}_size'])
            if not ok:
                rule_violations.append((r, f'lag_{lag}', msg))
                
    print(f"File: {fname} -> Total ML_Ready Rule Violations: {len(rule_violations)}")
    wb.close()
