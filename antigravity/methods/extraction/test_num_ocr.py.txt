import cv2
import asyncio
from PIL import Image
import winocr
import numpy as np

img = cv2.imread(r'extraction\extracted_zip\WhatsApp Image 2026-10-23 at 09.33.27.jpeg')

async def main():
    for k in range(10):
        yc = int(617 + k * 76.6)
        num_crop = img[yc-25:yc+25, 340:400]
        padded = cv2.copyMakeBorder(num_crop, 30, 30, 30, 30, cv2.BORDER_CONSTANT, value=[255, 255, 255])
        pil_im = Image.fromarray(cv2.cvtColor(padded, cv2.COLOR_BGR2RGB))
        res = await winocr.recognize_pil(pil_im)
        print(f"Row {k} padded OCR: '{res.text.strip()}'")

asyncio.run(main())
