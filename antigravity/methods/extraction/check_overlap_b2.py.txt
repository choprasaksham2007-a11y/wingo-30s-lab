import openpyxl, json

excel_path = r"project_reports\Colour_Prediction_Dataset_corrected.xlsx"
wb = openpyxl.load_workbook(excel_path, data_only=True)
ws = wb["Raw_Data"]

raw_data = {}
for r in ws.iter_rows(min_row=2, values_only=True):
    raw_data[str(r[1])] = {
        "seq": r[0],
        "pid": str(r[1]),
        "num": r[3],
        "col": r[4],
        "size": r[5]
    }

with open(r"extraction\reconciled_batch2.json") as f:
    b2_data = json.load(f)

overlap_pids = sorted([p for p in b2_data if p in raw_data])
print(f"Total overlap rounds: {len(overlap_pids)}")
mismatches = []
for p in overlap_pids:
    r = raw_data[p]
    b = b2_data[p]
    num_match = (r["num"] == b["number"])
    col_match = (r["col"] == b["colour"])
    sz_match = (r["size"] == b["size"])
    print(f"PID {p}: Old=(Num {r['num']}, {r['col']}, {r['size']}) vs New=(Num {b['number']}, {b['colour']}, {b['size']}) | Matches: {num_match and col_match and sz_match}")
    if not (num_match and col_match and sz_match):
        mismatches.append(p)

print(f"Total overlap mismatches: {len(mismatches)}")
