"""
Script to create Colour_Prediction_Dataset_corrected.xlsx.
Keeps original Colour_Prediction_Dataset.xlsx completely unchanged.
Adds 'colour_original' and 'fix_note' columns to Raw_Data.
Corrects the 26 rows in Raw_Data and updates ML_Ready, Summary, and Audit sheets accordingly.
"""
import os
import shutil
import json
import csv
import openpyxl
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils import get_column_letter
from collections import Counter
import sys

sys.path.append(r"extraction")
from src.ml_dataset_builder import is_consecutive_period

HEADER_FILL = PatternFill(start_color="1F497D", end_color="1F497D", fill_type="solid")
HEADER_FONT = Font(name="Calibri", size=11, bold=True, color="FFFFFF")
TITLE_FONT = Font(name="Calibri", size=14, bold=True, color="1F497D")
SECTION_FONT = Font(name="Calibri", size=12, bold=True, color="1F497D")
SUBTITLE_FONT = Font(name="Calibri", size=10, italic=True, color="595959")
REGULAR_FONT = Font(name="Calibri", size=10)
BOLD_FONT = Font(name="Calibri", size=10, bold=True)

CORRECTED_FILL = PatternFill(start_color="FFF2CC", end_color="FFF2CC", fill_type="solid") # Light yellow highlight

THIN_BORDER = Border(
    left=Side(style='thin', color='D9D9D9'),
    right=Side(style='thin', color='D9D9D9'),
    top=Side(style='thin', color='D9D9D9'),
    bottom=Side(style='thin', color='D9D9D9')
)

# Load changelog of 26 rows
with open(r"extraction\changelog_26.json") as f:
    changelog_list = json.load(f)
changelog_map = {item["sequence_no"]: item for item in changelog_list}

def create_corrected_workbook(original_excel_path: str, output_excel_path: str):
    wb_orig = openpyxl.load_workbook(original_excel_path, data_only=True)
    
    wb_new = openpyxl.Workbook()
    wb_new.remove(wb_new.active)
    
    # -------------------------------------------------------------
    # 1. SHEET: Raw_Data
    # -------------------------------------------------------------
    print("Generating corrected Raw_Data sheet...")
    ws_raw_orig = wb_orig["Raw_Data"]
    ws_raw_new = wb_new.create_sheet(title="Raw_Data")
    
    # Headers: sequence_no, Period ID, timestamp, number, colour, size, colour_original, fix_note
    headers = ["sequence_no", "Period ID", "timestamp", "number", "colour", "size", "colour_original", "fix_note"]
    ws_raw_new.append(headers)
    for c_idx in range(1, len(headers) + 1):
        cell = ws_raw_new.cell(row=1, column=c_idx)
        cell.fill = HEADER_FILL
        cell.font = HEADER_FONT
        cell.alignment = Alignment(horizontal="center", vertical="center")
        
    corrected_count = 0
    raw_records_for_ml = []
    
    for row in ws_raw_orig.iter_rows(min_row=2, values_only=True):
        seq = int(row[0])
        pid = str(row[1])
        ts = str(row[2])
        num = int(row[3])
        orig_col = str(row[4])
        sz = str(row[5])
        
        if seq in changelog_map:
            corrected_count += 1
            fix_info = changelog_map[seq]
            final_col = fix_info["new_colour"]
            note = fix_info["fix_note"]
            is_corrected = True
        else:
            final_col = orig_col
            note = "Original extraction verified"
            is_corrected = False
            
        row_vals = [seq, pid, ts, num, final_col, sz, orig_col, note]
        ws_raw_new.append(row_vals)
        cur_row = ws_raw_new.max_row
        
        # Enforce text data type on Period ID
        pid_cell = ws_raw_new.cell(row=cur_row, column=2)
        pid_cell.number_format = '@'
        pid_cell.data_type = 's'
        pid_cell.alignment = Alignment(horizontal="center")
        
        for c in range(1, len(headers) + 1):
            cell = ws_raw_new.cell(row=cur_row, column=c)
            cell.font = REGULAR_FONT
            cell.border = THIN_BORDER
            if c in [1, 2, 3, 4, 5, 6, 7]:
                cell.alignment = Alignment(horizontal="center")
            else:
                cell.alignment = Alignment(horizontal="left")
                
        # Highlight corrected rows
        if is_corrected:
            ws_raw_new.cell(row=cur_row, column=5).fill = CORRECTED_FILL # new colour
            ws_raw_new.cell(row=cur_row, column=7).fill = CORRECTED_FILL # colour_original
            ws_raw_new.cell(row=cur_row, column=8).fill = CORRECTED_FILL # fix_note
            
        raw_records_for_ml.append({
            "seq": seq, "period_id": pid, "timestamp": ts,
            "number": num, "colour": final_col, "size": sz
        })
        
    _autofit(ws_raw_new)
    print(f"Corrected {corrected_count} rows in Raw_Data.")

    # -------------------------------------------------------------
    # 2. SHEET: ML_Ready (Synchronized with corrected colours)
    # -------------------------------------------------------------
    print("Generating synchronized ML_Ready sheet...")
    ws_ml_new = wb_new.create_sheet(title="ML_Ready")
    
    ml_headers = [
        "sample_sequence", "Target Period ID", "target_timestamp",
        "prev_1_number", "prev_1_colour", "prev_1_size",
        "prev_2_number", "prev_2_colour", "prev_2_size",
        "prev_3_number", "prev_3_colour", "prev_3_size",
        "prev_4_number", "prev_4_colour", "prev_4_size",
        "prev_5_number", "prev_5_colour", "prev_5_size",
        "target_number", "target_colour", "target_size"
    ]
    ws_ml_new.append(ml_headers)
    for c_idx in range(1, len(ml_headers) + 1):
        cell = ws_ml_new.cell(row=1, column=c_idx)
        cell.fill = HEADER_FILL
        cell.font = HEADER_FONT
        cell.alignment = Alignment(horizontal="center", vertical="center")
        
    ml_sample_idx = 1
    corrected_ml_rows = []
    
    for i in range(5, len(raw_records_for_ml)):
        target_rec = raw_records_for_ml[i]
        lags = [raw_records_for_ml[i - k] for k in range(1, 6)]
        
        # Check window continuity
        window = [raw_records_for_ml[i - k] for k in range(5, -1, -1)]
        has_gap = False
        for w_idx in range(len(window) - 1):
            if not is_consecutive_period(window[w_idx]["period_id"], window[w_idx + 1]["period_id"]):
                has_gap = True
                break
        if has_gap:
            continue
            
        row_vals = [
            ml_sample_idx,
            str(target_rec["period_id"]),
            target_rec["timestamp"],
            lags[0]["number"], lags[0]["colour"], lags[0]["size"],
            lags[1]["number"], lags[1]["colour"], lags[1]["size"],
            lags[2]["number"], lags[2]["colour"], lags[2]["size"],
            lags[3]["number"], lags[3]["colour"], lags[3]["size"],
            lags[4]["number"], lags[4]["colour"], lags[4]["size"],
            target_rec["number"], target_rec["colour"], target_rec["size"]
        ]
        ws_ml_new.append(row_vals)
        cur_row = ws_ml_new.max_row
        
        pid_cell = ws_ml_new.cell(row=cur_row, column=2)
        pid_cell.number_format = '@'
        pid_cell.data_type = 's'
        pid_cell.alignment = Alignment(horizontal="center")
        
        for c in range(1, len(ml_headers) + 1):
            cell = ws_ml_new.cell(row=cur_row, column=c)
            cell.font = REGULAR_FONT
            cell.border = THIN_BORDER
            cell.alignment = Alignment(horizontal="center")
            
        corrected_ml_rows.append(row_vals)
        ml_sample_idx += 1
        
    _autofit(ws_ml_new)
    print(f"Generated {len(corrected_ml_rows)} synchronized ML samples.")

    # -------------------------------------------------------------
    # 3. SHEET: Read_Me
    # -------------------------------------------------------------
    ws_readme = wb_new.create_sheet(title="Read_Me")
    _populate_corrected_readme(ws_readme, len(raw_records_for_ml), len(corrected_ml_rows), corrected_count)

    # -------------------------------------------------------------
    # 4. SHEET: Summary
    # -------------------------------------------------------------
    ws_summary = wb_new.create_sheet(title="Summary")
    _populate_corrected_summary(ws_summary, raw_records_for_ml, len(corrected_ml_rows), corrected_count)

    # -------------------------------------------------------------
    # 5. SHEET: Missing_Rounds
    # -------------------------------------------------------------
    ws_missing = wb_new.create_sheet(title="Missing_Rounds")
    _copy_sheet_content(wb_orig["Missing_Rounds"], ws_missing)

    # -------------------------------------------------------------
    # 6. SHEET: Audit
    # -------------------------------------------------------------
    ws_audit = wb_new.create_sheet(title="Audit")
    _populate_corrected_audit(ws_audit, changelog_list)

    # -------------------------------------------------------------
    # 7. SHEET: User_Provided
    # -------------------------------------------------------------
    ws_user = wb_new.create_sheet(title="User_Provided")
    _copy_sheet_content(wb_orig["User_Provided"], ws_user)

    # -------------------------------------------------------------
    # 8. SHEET: ID_Audit
    # -------------------------------------------------------------
    ws_id = wb_new.create_sheet(title="ID_Audit")
    _copy_sheet_content(wb_orig["ID_Audit"], ws_id)

    wb_new.save(output_excel_path)
    print(f"[SUCCESS] Colour_Prediction_Dataset_corrected.xlsx saved successfully at: {output_excel_path}")
    
    return raw_records_for_ml, corrected_ml_rows

    ws.column_dimensions['B'].width = 85
    headers = ["Topic", "Details"]
    ws.append(headers)
    for c in [1, 2]:
        cell = ws.cell(row=1, column=c)
        cell.fill = HEADER_FILL
        cell.font = HEADER_FONT

    rows = [
        ("File Status", f"CORRECTED WORKBOOK: Exactly {corrected_count} rows in Raw_Data were corrected for colour dot overlay contamination. Original workbook remains completely intact."),
        ("Purpose", "Clean, chronological machine-learning historical dataset for 30-second colour-prediction games."),
        ("Confirmed Game Rules", "Number decides colour and size: 0 -> Red+Violet (Small); 1,3,7,9 -> Green; 2,4,6,8 -> Red; 5 -> Green+Violet (Big); 0-4 are Small, 5-9 are Big."),
        ("Cause of Error", "In the 26 corrected rows, floating interface icons (red lightning bolt badge and red/orange dragon overlay) covered the colour dot area in the screenshots, causing automated color masking to detect red pixels. Corrected using visible green number glyphs and confirmed rules."),
        ("New Columns in Raw_Data", "'colour_original' preserves the prior extracted value, and 'fix_note' documents the exact reason for modification."),
        ("Data Coverage", f"{total_rounds} total verified rounds sorted oldest to newest."),
        ("Chronological Range", "Oldest round: 20261022100051582 (2026-10-22 18:41:00) through 20261023100050438 (2026-10-23 09:09:00)."),
        ("30-Second Timestamps", "Continuous 30-second interval per period anchored to Period 50494 = 2026-10-23 09:37:00 AM."),
        ("ML_Ready Sheet", f"{total_ml} supervised samples with 5-round lag features predicting current round targets. Updated with corrected colours. Zero data leakage."),
        ("Missing_Rounds Sheet", "Documents the 5 page-jump gap intervals between screenshot batches with exact start/end timestamps."),
        ("User_Provided Sheet", "Tracks the 4 user-supplied numbers (51954=2, 51955=0, 51956=0, 51957=8).")
    ]
    for topic, detail in rows:
        ws.append([topic, detail])
        cur_row = ws.max_row
        ws.cell(row=cur_row, column=1).font = BOLD_FONT
        ws.cell(row=cur_row, column=1).border = THIN_BORDER
        c2 = ws.cell(row=cur_row, column=2)
        c2.font = REGULAR_FONT
        c2.border = THIN_BORDER
        c2.alignment = Alignment(wrap_text=True)


def _populate_corrected_summary(ws, records, total_ml, corrected_count):
    ws.column_dimensions['A'].width = 5
    ws.column_dimensions['B'].width = 44
    ws.column_dimensions['C'].width = 18
    ws.column_dimensions['D'].width = 5
    ws.column_dimensions['E'].width = 25
    ws.column_dimensions['F'].width = 18

    ws['B2'] = "COLOUR PREDICTION DATASET SUMMARY (CORRECTED)"
    ws['B2'].font = TITLE_FONT
    ws['B3'] = f"Updated Summary Reflecting {corrected_count} Colour Corrections (Floating Icon Occlusion Fixes)"
    ws['B3'].font = SUBTITLE_FONT

    ws['B5'] = "Dataset Overview"
    ws['B5'].font = SECTION_FONT
    
    kpis = [
        ("Total Verified Unique Rounds", len(records)),
        ("Total Eligible ML Training Samples", total_ml),
        ("Total Rows Corrected (Floating Icon Fix)", corrected_count),
        ("Unverified Records (Marked Unverified)", 0),
        ("Documented Gap Intervals (Missing Rounds)", 5),
        ("Directly Verified Records", sum(1 for r in records if r["seq"] not in [373, 374, 375, 376])),
        ("User-Supplied Outcomes", 4),
        ("Duplicate Overlaps Reconciled", 235),
        ("Conflicting Records Remaining", 0)
    ]
    cur_row = 6
    for label, val in kpis:
        ws.cell(row=cur_row, column=2, value=label).font = REGULAR_FONT
        ws.cell(row=cur_row, column=2).border = THIN_BORDER
        c_val = ws.cell(row=cur_row, column=3, value=val)
        c_val.font = BOLD_FONT
        c_val.alignment = Alignment(horizontal="right")
        c_val.border = THIN_BORDER
        cur_row += 1

    # Distribution by Number
    ws['E5'] = "Distribution by Number (0-9)"
    ws['E5'].font = SECTION_FONT
    num_counts = Counter([r["number"] for r in records if r["number"] is not None])
    
    ws.cell(row=6, column=5, value="Number").font = HEADER_FONT
    ws.cell(row=6, column=5).fill = HEADER_FILL
    ws.cell(row=6, column=6, value="Count").font = HEADER_FONT
    ws.cell(row=6, column=6).fill = HEADER_FILL
    
    r_idx = 7
    for digit in range(10):
        ws.cell(row=r_idx, column=5, value=f"Number {digit}").border = THIN_BORDER
        c = ws.cell(row=r_idx, column=6, value=num_counts.get(digit, 0))
        c.font = BOLD_FONT
        c.alignment = Alignment(horizontal="right")
        c.border = THIN_BORDER
        r_idx += 1

    # Distribution by Colour (Updated!)
    ws.cell(row=r_idx + 1, column=5, value="Distribution by Colour (Corrected)").font = SECTION_FONT
    col_counts = Counter([r["colour"] for r in records if r["colour"]])
    
    ws.cell(row=r_idx + 2, column=5, value="Colour").font = HEADER_FONT
    ws.cell(row=r_idx + 2, column=5).fill = HEADER_FILL
    ws.cell(row=r_idx + 2, column=6, value="Count").font = HEADER_FONT
    ws.cell(row=r_idx + 2, column=6).fill = HEADER_FILL
    
    r_col = r_idx + 3
    # Order: Green, Red, Green+Violet, Red+Violet
    for col_name in ["Green", "Red", "Green+Violet", "Red+Violet"]:
        ws.cell(row=r_col, column=5, value=col_name).border = THIN_BORDER
        c = ws.cell(row=r_col, column=6, value=col_counts.get(col_name, 0))
        c.font = BOLD_FONT
        c.alignment = Alignment(horizontal="right")
        c.border = THIN_BORDER
        r_col += 1


def _populate_corrected_audit(ws, changelog_list):
    ws.column_dimensions['A'].width = 14
    ws.column_dimensions['B'].width = 22
    ws.column_dimensions['C'].width = 10
    ws.column_dimensions['D'].width = 14
    ws.column_dimensions['E'].width = 16
    ws.column_dimensions['F'].width = 24
    ws.column_dimensions['G'].width = 75

    ws['A1'] = "PERIOD ID & COLOUR AUDIT REPORT - CORRECTION PASS"
    ws['A1'].font = TITLE_FONT
    ws['A2'] = "Independent two-pass verification, cross-reconciliation, and occlusion recovery."
    ws['A2'].font = SUBTITLE_FONT

    # 1. Audit Check Summary Table (with honest statement for Pass 2)
    audit_headers = ["Audit Check", "Result", "Details"]
    for c_idx, h in enumerate(audit_headers, start=1):
        cell = ws.cell(row=4, column=c_idx, value=h)
        cell.fill = HEADER_FILL
        cell.font = HEADER_FONT
        cell.alignment = Alignment(horizontal="center")

    audit_rows = [
        ("Pass 1: Record Extraction", "1,495 observations extracted", "Extracted all row items across 153 screenshots using line-aligned bounding boxes."),
        ("Pass 2: Independent Audit", "Numbers and sizes verified; 26 colour mismatches found and corrected from the number rule.", "Re-verified numbers against Big/Small labels and screenshots. 26 colour dots contaminated by floating badges on consecutive pages corrected to Green/Green+Violet based on visible number glyphs and confirmed rules."),
        ("Cross-Dataset Overlap", "22 rounds reconfirmed", "Rounds 52060 to 52081 cross-checked between previous dataset and new screenshots with 100% agreement."),
        ("Colour Occlusion Recovery", "77 rounds recovered from overlap", "Floating icons obscuring dots in one screenshot were recovered from overlapping views."),
        ("Rule-Derived Fallback", "Applied where floating icons occluded dots", "Where floating icons covered the dot in all available screenshots, colour was derived from verified number rulebook."),
        ("User-Supplied Outcomes", "4 manual entries documented", "Rounds 51954 to 51957 clearly flagged as User-Provided and rule-derived."),
        ("Sequential Continuity", "Documented in Missing_Rounds", "5 gap intervals between screenshot batches accurately tracked with timestamp boundaries."),
        ("Data Type Integrity", "100% exact text preservation", "Every Period ID stored strictly as text string (@) with zero precision truncation.")
    ]

    for r_idx, (chk, res, det) in enumerate(audit_rows, start=5):
        ws.cell(row=r_idx, column=1, value=chk).font = BOLD_FONT
        ws.cell(row=r_idx, column=1).border = THIN_BORDER
        c_res = ws.cell(row=r_idx, column=2, value=res)
        c_res.font = BOLD_FONT if "mismatches" in res else REGULAR_FONT
        c_res.border = THIN_BORDER
        c_det = ws.cell(row=r_idx, column=3, value=det)
        c_det.font = REGULAR_FONT
        c_det.border = THIN_BORDER

    # 2. Detailed Changelog Table for the 26 Corrected Rows
    start_r = 15
    ws.cell(row=start_r, column=1, value="26 Corrected Colour Rows (Occlusion Changelog)").font = SECTION_FONT
    
    cl_headers = ["sequence_no", "Period ID", "number", "old_colour", "new_colour", "verified_from_screenshot", "visual_evidence"]
    for c_idx, h in enumerate(cl_headers, start=1):
        cell = ws.cell(row=start_r + 1, column=c_idx, value=h)
        cell.fill = HEADER_FILL
        cell.font = HEADER_FONT
        cell.alignment = Alignment(horizontal="center")

    for r_idx, item in enumerate(changelog_list, start=start_r + 2):
        row_vals = [
            item["sequence_no"],
            str(item["period_id"]),
            item["number"],
            item["old_colour"],
            item["new_colour"],
            "Yes (Glyph & Rules)",
            item["visual_evidence"]
        ]
        for c_idx, val in enumerate(row_vals, start=1):
            cell = ws.cell(row=r_idx, column=c_idx, value=val)
            cell.font = REGULAR_FONT
            cell.border = THIN_BORDER
            if c_idx in [1, 3, 4, 5, 6]:
                cell.alignment = Alignment(horizontal="center")
            elif c_idx == 2:
                cell.number_format = '@'
                cell.data_type = 's'
                cell.alignment = Alignment(horizontal="center")
            else:
                cell.alignment = Alignment(horizontal="left")


def _copy_sheet_content(ws_src, ws_dst):
    for r in ws_src.iter_rows(values_only=True):
        ws_dst.append(list(r))
    for col in ws_dst.columns:
        max_len = 0
        col_letter = get_column_letter(col[0].column)
        for cell in col:
            val_str = str(cell.value or '')
            max_len = max(max_len, len(val_str))
        ws_dst.column_dimensions[col_letter].width = max(max_len + 4, 14)


def _autofit(ws):
    for col in ws.columns:
        max_len = 0
        col_letter = get_column_letter(col[0].column)
        for cell in col:
            val_str = str(cell.value or '')
            if '\n' in val_str:
                val_str = max(val_str.split('\n'), key=len)
            max_len = max(max_len, len(val_str))
        ws.column_dimensions[col_letter].width = max(max_len + 4, 14)

    raw_recs, ml_recs = create_corrected_workbook(orig_file, out_file)
