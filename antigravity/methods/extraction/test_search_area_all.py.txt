import cv2, os, glob, numpy as np, asyncio, winocr
from PIL import Image

batch3_dir = r"extraction\extracted_zip_batch3"
templates = np.load(r"extraction\digit_templates.npy", allow_pickle=True).item()

f = os.path.join(batch3_dir, "WhatsApp Image 2026-10-24 at 00.27.08.jpeg")
im = cv2.imread(f)

pid_strip = im[500:1450, 20:300]
res_pid = asyncio.run(winocr.recognize_pil(Image.fromarray(cv2.cvtColor(pid_strip, cv2.COLOR_BGR2RGB))))

bs_strip = im[500:1450, 420:530]
res_bs = asyncio.run(winocr.recognize_pil(Image.fromarray(cv2.cvtColor(bs_strip, cv2.COLOR_BGR2RGB))))
bs_items = []
for l in res_bs.lines:
    txt = " ".join([w.text.strip() for w in l.words])
    if "Big" in txt or "Small" in txt or "sma" in txt.lower():
        val = "Small" if ("Small" in txt or "sma" in txt.lower()) else "Big"
        y_c = l.words[0].bounding_rect.y + l.words[0].bounding_rect.height / 2.0
        bs_items.append((y_c, val))

for line in res_pid.lines:
    txt = ''.join([w.text.strip() for w in line.words]).replace(' ', '').replace('o', '0').replace('O', '0')
    if len(txt) >= 17 and '202610231000' in txt:
        pid = [txt[i:i+17] for i in range(len(txt)-16) if txt[i:i+17].startswith('202610231000')][0]
        w0 = line.words[0]
        y_rel = w0.bounding_rect.y + w0.bounding_rect.height / 2.0
        yc = int(round(500 + y_rel))
        
        # Search area: y from yc-20 to yc+20, x from 315 to 375
        search_area = im[yc-20:yc+20, 315:375]
        
        scores = {}
        for d, tpl in templates.items():
            res = cv2.matchTemplate(search_area, tpl, cv2.TM_CCOEFF_NORMED)
            scores[int(d)] = round(float(cv2.minMaxLoc(res)[1]), 3)
            
        # Color mask
        hsv = cv2.cvtColor(search_area, cv2.COLOR_BGR2HSV)
        g_cnt = cv2.countNonZero(cv2.inRange(hsv, np.array([35, 50, 50]), np.array([85, 255, 255])))
        r_cnt = cv2.countNonZero(cv2.inRange(hsv, np.array([0, 70, 70]), np.array([10, 255, 255])) | cv2.inRange(hsv, np.array([170, 70, 70]), np.array([180, 255, 255])))
        v_cnt = cv2.countNonZero(cv2.inRange(hsv, np.array([125, 40, 40]), np.array([165, 255, 255])))
        
        # Match size
        best_bs = None
        min_d = 999.0
        for b_y, b_val in bs_items:
            dist = abs(b_y - y_rel)
            if dist < min_d and dist < 35:
                min_d = dist
                best_bs = b_val
                
        # Candidate filter based on color & size:
        if v_cnt > 60:
            if g_cnt > 60:
                best_d = 5
            else:
                best_d = 0
        elif g_cnt > r_cnt:
            # Green: 1, 3 (Small) or 7, 9 (Big)
            if best_bs == "Small":
                cands = [1, 3]
            elif best_bs == "Big":
                cands = [7, 9]
            else:
                cands = [1, 3, 7, 9]
            best_d = max(cands, key=lambda d: scores[d])
        else:
            # Red: 2, 4 (Small) or 6, 8 (Big)
            if best_bs == "Small":
                cands = [2, 4]
            elif best_bs == "Big":
                cands = [6, 8]
            else:
                cands = [2, 4, 6, 8]
            best_d = max(cands, key=lambda d: scores[d])
            
        print(f"PID {pid} at yc={yc}: Best Digit={best_d} (Score={scores[best_d]}) | BS={best_bs} | G={g_cnt}, R={r_cnt}, V={v_cnt}")
