import cv2
import asyncio
from PIL import Image
import winocr

img = cv2.imread(r"extraction\extracted_zip\WhatsApp Image 2026-10-23 at 09.33.27.jpeg")

async def main():
    crop = img[500:1400, 320:570]
    pil_im = Image.fromarray(cv2.cvtColor(crop, cv2.COLOR_BGR2RGB))
    res = await winocr.recognize_pil(pil_im)
    print("Number + BigSmall crop OCR lines:")
    for line in res.lines:
        line_txt = " ".join([w.text for w in line.words])
        y = line.words[0].bounding_rect.y
        print(f"y={y:.1f}: {line_txt}")

asyncio.run(main())
