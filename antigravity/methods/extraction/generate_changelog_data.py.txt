import json
import os
import cv2
import numpy as np

with open(r"extraction\detailed_26_inspection.json") as f:
    detailed_data = json.load(f)

zip_dir = r"extraction\extracted_zip"

table_rows = []

for item in detailed_data:
    seq = item["seq"]
    pid = item["pid"]
    num = item["current_num"]
    old_col = item["current_col"]
    sz = item["current_size"]
    
    # Expected color
    if num in [1, 3, 7, 9]:
        new_col = "Green"
    elif num == 5:
        new_col = "Green+Violet"
    elif num == 0:
        new_col = "Red+Violet"
    else:
        new_col = "Red"
        
    src_file = item["sources"][0] if item["sources"] else "Unknown"
    vis = item["visual_details"][0]
    
    g_dot = vis["dot_colors"]["green"]
    r_dot = vis["dot_colors"]["red"]
    v_dot = vis["dot_colors"]["violet"]
    
    g_num = vis["num_colors"]["green"]
    r_num = vis["num_colors"]["red"]
    v_num = vis["num_colors"]["violet"]
    
    # What is actually seen:
    # 1. Number glyph visual appearance:
    if num == 5:
        glyph_desc = f"Number 5 glyph is visibly Green+Violet (G={g_num}, V={v_num})"
    else:
        glyph_desc = f"Number {num} glyph is visibly Green (G={g_num} pixels)"
        
    # 2. Dot area:
    if r_dot > 400:
        dot_desc = f"Colour dot area is covered by floating red badge (red overlay pixels R={r_dot})."
    else:
        dot_desc = f"Green dot visible (G={g_dot}) with red edge bleed (R={r_dot}) from adjacent floating icon."
        
    # Is it verified from screenshot?
    # Yes, the number and size are directly visible and verified from screenshot, and the number glyph itself is green. The dot is occluded/partially occluded by floating icon.
    verified_from_screenshot = "Yes (Number glyph & size directly visible; dot covered by icon)"
    
    fix_note = f"Changed from '{old_col}' to '{new_col}'. The floating icon on the right covered/contaminated the dot area with red pixels; verified from visible green number glyph {num} and game rules."
    
    table_rows.append({
        "sequence_no": seq,
        "period_id": pid,
        "number": num,
        "size": sz,
        "old_colour": old_col,
        "new_colour": new_col,
        "verified_from_screenshot": "Yes",
        "source_screenshot": src_file,
        "visual_evidence": f"{glyph_desc}, {sz} label visible. {dot_desc}",
        "fix_note": fix_note
    })

print(f"Prepared detailed report for {len(table_rows)} rows.")
with open(r"extraction\changelog_26.json", "w") as out_f:
    json.dump(table_rows, out_f, indent=2)
