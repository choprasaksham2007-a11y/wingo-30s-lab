import cv2
import numpy as np

templates = np.load(r"extraction\digit_templates.npy", allow_pickle=True).item()
print("Loaded templates for digits:", sorted(list(templates.keys())))

# Let's write a robust classifier function
def classify_digit(num_crop):
    best_digit = None
    best_score = -1.0
    for digit, tpl in templates.items():
        # Match template using normalized cross correlation
        res = cv2.matchTemplate(num_crop, tpl, cv2.TM_CCOEFF_NORMED)
        _, max_val, _, _ = cv2.minMaxLoc(res)
        if max_val > best_score:
            best_score = max_val
            best_digit = digit
    return best_digit, best_score

# Test on WhatsApp Image 2026-10-23 at 09.33.27.jpeg (known: 4, 5, 4, 1, 2, 4, 4, 2, 5, 1)
img = cv2.imread(r"extraction\extracted_zip\WhatsApp Image 2026-10-23 at 09.33.27.jpeg")
print("Testing classifier on known image:")
for k in range(10):
    yc = int(617 + k * 76.6)
    num_crop = img[yc-20:yc+20, 345:385]
    d, s = classify_digit(num_crop)
    print(f"Row {k}: Classified as {d} with score={s:.4f}")
