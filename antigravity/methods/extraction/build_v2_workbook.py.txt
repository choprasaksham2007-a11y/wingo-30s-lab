"""
Build Colour_Prediction_Dataset_v2.xlsx merging:
  - Existing 1,260 corrected rounds (from Colour_Prediction_Dataset_corrected.xlsx)
  - 380 new rounds from batch 2 screenshots (PIDs 50439-50818)
  - 2 number corrections discovered via cross-verification: PID 50430 (8->6), PID 50431 (8->6)
Total: 1,640 rounds.
"""
import os, sys, shutil, json
import openpyxl
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils import get_column_letter
from collections import Counter
from datetime import datetime, timedelta

sys.path.append(r"extraction")
from src.ml_dataset_builder import is_consecutive_period

# ── Styling ──
HEADER_FILL = PatternFill(start_color="1F497D", end_color="1F497D", fill_type="solid")
HEADER_FONT = Font(name="Calibri", size=11, bold=True, color="FFFFFF")
TITLE_FONT  = Font(name="Calibri", size=14, bold=True, color="1F497D")
SECTION_FONT = Font(name="Calibri", size=12, bold=True, color="1F497D")
SUBTITLE_FONT = Font(name="Calibri", size=10, italic=True, color="595959")
REGULAR_FONT = Font(name="Calibri", size=10)
BOLD_FONT    = Font(name="Calibri", size=10, bold=True)
CORRECTED_FILL = PatternFill(start_color="FFF2CC", end_color="FFF2CC", fill_type="solid")
NEW_FILL = PatternFill(start_color="E2EFDA", end_color="E2EFDA", fill_type="solid")
THIN_BORDER = Border(
    left=Side(style='thin', color='D9D9D9'),
    right=Side(style='thin', color='D9D9D9'),
    top=Side(style='thin', color='D9D9D9'),
    bottom=Side(style='thin', color='D9D9D9')
)

def get_timestamp(pid: str) -> str:
    anchor = datetime(2026, 10, 23, 9, 37, 0)
    date_part = pid[:8]
    suffix = int(pid[-5:])
    if date_part == "20261023":
        diff = suffix - 50494
        dt = anchor + timedelta(seconds=diff * 30)
    elif date_part == "20261022":
        dt_50001 = anchor - timedelta(seconds=(50494 - 50001) * 30)
        dt_52880 = dt_50001 - timedelta(seconds=30)
        diff_22 = 52880 - suffix
        dt = dt_52880 - timedelta(seconds=diff_22 * 30)
    else:
        return ""
    return dt.strftime("%Y-%m-%d %H:%M:%S")

def derive_colour(n):
    if n == 0: return "Red+Violet"
    if n == 5: return "Green+Violet"
    if n in [1,3,7,9]: return "Green"
    return "Red"

def derive_size(n):
    return "Small" if n <= 4 else "Big"

# ── Load existing corrected data ──
corr_path = r"project_reports\Colour_Prediction_Dataset_corrected.xlsx"
wb_old = openpyxl.load_workbook(corr_path, data_only=True)
ws_old = wb_old["Raw_Data"]

existing_records = []
for row in ws_old.iter_rows(min_row=2, values_only=True):
    existing_records.append({
        "seq": int(row[0]),
        "pid": str(row[1]),
        "ts": str(row[2]),
        "num": int(row[3]),
        "col": str(row[4]),
        "size": str(row[5]),
        "col_orig": str(row[6]),
        "fix_note": str(row[7])
    })

print(f"Loaded {len(existing_records)} existing records.")

# ── Apply 2 number corrections from cross-verification ──
num_corrections = {
    "20261023100050430": {"old_num": 8, "new_num": 6, "note": "Cross-verified with batch2 screenshots: template score(6)=0.80 vs score(8)=0.61; Size OCR='Big' matches both; corrected 8->6."},
    "20261023100050431": {"old_num": 8, "new_num": 6, "note": "Cross-verified with batch2 screenshots: template score(6)=0.80 vs score(8)=0.65; Size OCR='Big' matches both; corrected 8->6."}
}

num_change_count = 0
for rec in existing_records:
    if rec["pid"] in num_corrections:
        fix = num_corrections[rec["pid"]]
        old_n = rec["num"]
        new_n = fix["new_num"]
        rec["num"] = new_n
        rec["col"] = derive_colour(new_n)
        rec["size"] = derive_size(new_n)
        rec["col_orig"] = rec["col_orig"]  # keep original colour_original
        rec["fix_note"] = fix["note"]
        num_change_count += 1
        print(f"  Corrected PID {rec['pid']}: number {old_n}->{new_n}, colour->Red, size->Big")

# ── Load batch 2 new rounds ──
with open(r"extraction\reconciled_batch2_refined.json") as f:
    b2_data = json.load(f)

existing_pids = set(r["pid"] for r in existing_records)
new_pids = sorted([p for p in b2_data if p not in existing_pids])
print(f"New unique rounds from batch 2: {len(new_pids)}")

new_records = []
for pid in new_pids:
    b2 = b2_data[pid]
    new_records.append({
        "seq": 0,  # will be re-sequenced
        "pid": pid,
        "ts": get_timestamp(pid),
        "num": b2["number"],
        "col": derive_colour(b2["number"]),
        "size": derive_size(b2["number"]),
        "col_orig": derive_colour(b2["number"]),
        "fix_note": "Extracted from batch 2 screenshots"
    })

# ── Merge and re-sequence ──
all_records = existing_records + new_records
# Sort chronologically
all_records.sort(key=lambda r: (r["pid"][:8], int(r["pid"][-5:])))
for i, rec in enumerate(all_records):
    rec["seq"] = i + 1

total_rows = len(all_records)
print(f"Total merged records: {total_rows}")

# ── Identify gaps for Missing_Rounds ──
gaps = []
for i in range(len(all_records) - 1):
    p1 = all_records[i]["pid"]
    p2 = all_records[i+1]["pid"]
    if not is_consecutive_period(p1, p2):
        d1, d2 = p1[:8], p2[:8]
        s1, s2 = int(p1[-5:]), int(p2[-5:])
        if d1 == d2:
            missing = s2 - s1 - 1
        elif d1 == "20261022" and d2 == "20261023":
            missing = (52880 - s1) + (s2 - 50001)
        else:
            missing = 0
        gaps.append({
            "gap_num": len(gaps) + 1,
            "before_pid": p1,
            "after_pid": p2,
            "before_seq": all_records[i]["seq"],
            "after_seq": all_records[i+1]["seq"],
            "missing_count": missing,
            "before_ts": all_records[i]["ts"],
            "after_ts": all_records[i+1]["ts"]
        })

print(f"Total gaps: {len(gaps)}")
for g in gaps:
    print(f"  Gap {g['gap_num']}: {g['before_pid']} -> {g['after_pid']} ({g['missing_count']} missing)")

# ── Build ML_Ready ──
ml_rows = []
ml_seq = 1
for i in range(5, len(all_records)):
    target = all_records[i]
    window = [all_records[i-k] for k in range(5, -1, -1)]
    has_gap = False
    for w in range(len(window)-1):
        if not is_consecutive_period(window[w]["pid"], window[w+1]["pid"]):
            has_gap = True
            break
    if has_gap:
        continue
    lags = [all_records[i-k] for k in range(1, 6)]
    ml_rows.append({
        "ml_seq": ml_seq,
        "target": target,
        "lags": lags
    })
    ml_seq += 1

print(f"ML_Ready rows: {len(ml_rows)}")

# ── Colour distribution ──
col_counts = Counter(r["col"] for r in all_records)
print(f"Colour distribution: {dict(col_counts)}")

# ────────────────────────────────────────────
# CREATE THE WORKBOOK
# ────────────────────────────────────────────

wb.remove(wb.active)

# ── 1. Raw_Data ──
print("Writing Raw_Data sheet...")
ws_raw = wb.create_sheet(title="Raw_Data")
headers = ["sequence_no", "Period ID", "timestamp", "number", "colour", "size", "colour_original", "fix_note"]
ws_raw.append(headers)
for c_idx in range(1, len(headers)+1):
    cell = ws_raw.cell(row=1, column=c_idx)
    cell.fill = HEADER_FILL
    cell.font = HEADER_FONT
    cell.alignment = Alignment(horizontal="center", vertical="center")

for rec in all_records:
    ws_raw.append([rec["seq"], rec["pid"], rec["ts"], rec["num"], rec["col"], rec["size"], rec["col_orig"], rec["fix_note"]])
    cur = ws_raw.max_row
    # Period ID as text
    pid_cell = ws_raw.cell(row=cur, column=2)
    pid_cell.number_format = '@'
    pid_cell.data_type = 's'
    for c in range(1, len(headers)+1):
        cell = ws_raw.cell(row=cur, column=c)
        cell.font = REGULAR_FONT
        cell.border = THIN_BORDER
        cell.alignment = Alignment(horizontal="center") if c <= 7 else Alignment(horizontal="left")
    # Highlight corrected or new rows
    if rec["pid"] in num_corrections:
        for c in [4, 5, 6, 8]:
            ws_raw.cell(row=cur, column=c).fill = CORRECTED_FILL
    elif rec["seq"] > len(existing_records) - num_change_count:
        pass  # new rows, no highlight needed

# Auto-fit columns
for col in ws_raw.columns:
    max_len = max(len(str(cell.value or '')) for cell in col)
    ws_raw.column_dimensions[get_column_letter(col[0].column)].width = max(max_len + 4, 14)
