import cv2
import asyncio
from PIL import Image
import winocr

img = cv2.imread(r'extraction\extracted_zip\WhatsApp Image 2026-10-23 at 09.33.27.jpeg')

async def main():
    # Crop entire table: x from 40 to 680, y from 500 to 1400
    table_crop = img[500:1400, 40:680]
    pil_im = Image.fromarray(cv2.cvtColor(table_crop, cv2.COLOR_BGR2RGB))
    res = await winocr.recognize_pil(pil_im)
    print("Table crop OCR:")
    for line in res.lines:
        line_text = " ".join([w.text for w in line.words])
        y = line.words[0].bounding_rect.y
        print(f"y={y:.1f}: {line_text}")

asyncio.run(main())
