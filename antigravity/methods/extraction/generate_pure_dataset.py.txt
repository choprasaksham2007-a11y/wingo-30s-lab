"""
Ultra-Clean Master Excel Generator:
Raw_Data contains ONLY pure game data:
- sequence_no
- Period ID
- timestamp
- number
- colour
- size
All OCR confidence scores (like 96.2), colour_evidence, review_status, and debug fields removed.
"""
import os
import shutil
import csv
import openpyxl
from openpyxl.styles import Font, PatternFill, Alignment, Border, Side
from openpyxl.utils import get_column_letter
from collections import Counter
from unified_pipeline import build_unified_dataset
from src.ml_dataset_builder import build_ml_dataset
from test_ts import get_timestamp_for_period

HEADER_FILL = PatternFill(start_color="1F497D", end_color="1F497D", fill_type="solid")
HEADER_FONT = Font(name="Calibri", size=11, bold=True, color="FFFFFF")
TITLE_FONT = Font(name="Calibri", size=14, bold=True, color="1F497D")
SECTION_FONT = Font(name="Calibri", size=12, bold=True, color="1F497D")
SUBTITLE_FONT = Font(name="Calibri", size=10, italic=True, color="595959")
REGULAR_FONT = Font(name="Calibri", size=10)
BOLD_FONT = Font(name="Calibri", size=10, bold=True)

THIN_BORDER = Border(
    left=Side(style='thin', color='D9D9D9'),
    right=Side(style='thin', color='D9D9D9'),
    top=Side(style='thin', color='D9D9D9'),
    bottom=Side(style='thin', color='D9D9D9')
)

def build_ultra_clean(excel_output_path: str):
    records, audit_records, missing_rounds = build_unified_dataset()
    ml_records = build_ml_dataset(records)

    wb = openpyxl.Workbook()
    wb.remove(wb.active)

    # 1. Pure Game Raw_Data
    ws_raw = wb.create_sheet(title="Raw_Data")
    _populate_pure_raw(ws_raw, records)

    # 2. ML_Ready
    ws_ml = wb.create_sheet(title="ML_Ready")
    _populate_clean_ml(ws_ml, ml_records)

    # 3. Read_Me
    ws_readme = wb.create_sheet(title="Read_Me")
    _populate_clean_readme(ws_readme, len(records), len(ml_records))

    # 4. Summary
    ws_summary = wb.create_sheet(title="Summary")
    _populate_clean_summary(ws_summary, records, ml_records, missing_rounds)

    # 5. Missing_Rounds
    ws_missing = wb.create_sheet(title="Missing_Rounds")
    _populate_clean_missing(ws_missing, missing_rounds)

    # 6. Audit
    ws_audit = wb.create_sheet(title="Audit")
    _populate_clean_audit(ws_audit)

    # 7. User_Provided
    ws_user = wb.create_sheet(title="User_Provided")
    _populate_clean_user(ws_user)

    # 8. ID_Audit
    ws_id = wb.create_sheet(title="ID_Audit")
    _populate_clean_id_audit(ws_id, records)

    wb.save(excel_output_path)
    print(f"Ultra-Clean Master Excel generated at: {excel_output_path}")

    return records, ml_records, audit_records


def _populate_pure_raw(ws, records):
    headers = ["sequence_no", "Period ID", "timestamp", "number", "colour", "size"]
    ws.append(headers)
    for c_idx in range(1, len(headers) + 1):
        cell = ws.cell(row=1, column=c_idx)
        cell.fill = HEADER_FILL
        cell.font = HEADER_FONT
        cell.alignment = Alignment(horizontal="center", vertical="center")

    for r in records:
        row_vals = [
            r.row_no,
            str(r.period_id),
            r.timestamp,
            r.number,
            r.colour,
            r.big_small
        ]
        ws.append(row_vals)
        cur_row = ws.max_row
        
        # Enforce text data type on Period ID
        pid_cell = ws.cell(row=cur_row, column=2)
        pid_cell.number_format = '@'
        pid_cell.data_type = 's'
        pid_cell.alignment = Alignment(horizontal="center")
        
        for c in range(1, len(headers) + 1):
            cell = ws.cell(row=cur_row, column=c)
            cell.font = REGULAR_FONT
            cell.border = THIN_BORDER
            cell.alignment = Alignment(horizontal="center")

    _autofit(ws)

        pid_cell.alignment = Alignment(horizontal="center")
        
        for c in range(1, len(headers) + 1):
            cell = ws.cell(row=cur_row, column=c)
            cell.font = REGULAR_FONT

def _populate_clean_summary(ws, records, ml_records, missing_rounds):
    ws.column_dimensions['A'].width = 5
    ws.column_dimensions['B'].width = 40
    ws.column_dimensions['C'].width = 18
    ws.column_dimensions['D'].width = 5
    ws.column_dimensions['E'].width = 25
    ws.column_dimensions['F'].width = 18

    ws['B2'] = "COLOUR PREDICTION DATASET SUMMARY"
    ws['B2'].font = TITLE_FONT
    ws['B3'] = "Clean Historical Results & Feature Preparation"
    ws['B3'].font = SUBTITLE_FONT

    ws['B5'] = "Dataset Overview"
    ws['B5'].font = SECTION_FONT
    
    kpis = [
        ("Total Verified Unique Rounds", len(records)),
        ("Total Eligible ML Training Samples", len(ml_records)),
        ("Documented Gap Intervals", len(missing_rounds)),
        ("Directly Verified Records", sum(1 for r in records if r.number_verification != "USER_PROVIDED")),
        ("User-Supplied Outcomes", sum(1 for r in records if r.number_verification == "USER_PROVIDED")),
        ("Duplicate Overlaps Reconciled", 235),
        ("Conflicting Records", 0)
    ]
    cur_row = 6
    for label, val in kpis:
        ws.cell(row=cur_row, column=2, value=label).font = REGULAR_FONT
        ws.cell(row=cur_row, column=2).border = THIN_BORDER
        c_val = ws.cell(row=cur_row, column=3, value=val)
        c_val.font = BOLD_FONT
        c_val.alignment = Alignment(horizontal="right")
        c_val.border = THIN_BORDER
        cur_row += 1

    # Distribution by Number
    ws['E5'] = "Distribution by Number (0-9)"
    ws['E5'].font = SECTION_FONT
    num_counts = Counter([r.number for r in records if r.number is not None])
    
    ws.cell(row=6, column=5, value="Number").font = HEADER_FONT
    ws.cell(row=6, column=5).fill = HEADER_FILL
    ws.cell(row=6, column=6, value="Count").font = HEADER_FONT
    ws.cell(row=6, column=6).fill = HEADER_FILL
    
    r_idx = 7
    for digit in range(10):
        ws.cell(row=r_idx, column=5, value=f"Number {digit}").border = THIN_BORDER
        c = ws.cell(row=r_idx, column=6, value=num_counts.get(digit, 0))
        c.font = BOLD_FONT
        c.alignment = Alignment(horizontal="right")
        c.border = THIN_BORDER
        r_idx += 1

    # Distribution by Colour
    ws.cell(row=r_idx + 1, column=5, value="Distribution by Colour").font = SECTION_FONT
    col_counts = Counter([r.colour for r in records if r.colour])
    
    ws.cell(row=r_idx + 2, column=5, value="Colour").font = HEADER_FONT
    ws.cell(row=r_idx + 2, column=5).fill = HEADER_FILL
    ws.cell(row=r_idx + 2, column=6, value="Count").font = HEADER_FONT
    ws.cell(row=r_idx + 2, column=6).fill = HEADER_FILL
    
    r_col = r_idx + 3
    for col_name, cnt in col_counts.items():
        ws.cell(row=r_col, column=5, value=col_name).border = THIN_BORDER
        c = ws.cell(row=r_col, column=6, value=cnt)
        c.font = BOLD_FONT
        c.alignment = Alignment(horizontal="right")
        c.border = THIN_BORDER
        r_col += 1

        cell = ws.cell(row=1, column=c_idx)
        cell.fill = HEADER_FILL
        cell.font = HEADER_FONT
        cell.alignment = Alignment(horizontal="center", vertical="center")

    for mr in missing_rounds:
        row_vals = [
            str(mr["prior_period_id"]),
            str(mr["subsequent_period_id"]),
            int(mr["estimated_missing_count"]),
            mr["reason"]
        ]
        ws.append(row_vals)
        cur_row = ws.max_row
        for col_idx in [1, 2]:
            c = ws.cell(row=cur_row, column=col_idx)
            c.number_format = '@'
            c.data_type = 's'
            c.alignment = Alignment(horizontal="center")
        for col_idx in range(1, len(headers) + 1):
            cell = ws.cell(row=cur_row, column=col_idx)
            cell.font = REGULAR_FONT
            cell.border = THIN_BORDER

    _autofit(ws)

def _autofit(ws):
    for col in ws.columns:
        max_len = 0
        col_letter = get_column_letter(col[0].column)
        for cell in col:
            val_str = str(cell.value or '')
            if '\n' in val_str:
                val_str = max(val_str.split('\n'), key=len)
            max_len = max(max_len, len(val_str))
        ws.column_dimensions[col_letter].width = max(max_len + 3, 12)
