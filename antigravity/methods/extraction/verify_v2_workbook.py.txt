import openpyxl, sys
from collections import Counter

sys.path.append(r"extraction")
from src.ml_dataset_builder import is_consecutive_period

file_path = r"project_reports\Colour_Prediction_Dataset_v2.xlsx"
wb = openpyxl.load_workbook(file_path, data_only=True)

print("Sheet names:", wb.sheetnames)

# 1. Raw_Data checks
ws_raw = wb["Raw_Data"]
raw_rows = list(ws_raw.iter_rows(min_row=2, values_only=False))
print(f"Total Raw_Data rows: {len(raw_rows)} (Expected: 1640)")

pids = [r[1].value for r in raw_rows]
pid_types = [r[1].data_type for r in raw_rows]
print("All PIDs unique:", len(set(pids)) == len(pids))
print("All PIDs 17 digits:", all(len(str(p)) == 17 and str(p).isdigit() for p in pids))
print("All PIDs stored as text string ('s'):", all(t == 's' for t in pid_types))

# Check rules
rule_violations = []
for r in raw_rows:
    seq = r[0].value
    pid = r[1].value
    num = r[3].value
    col = r[4].value
    sz = r[5].value
    
    exp_sz = "Small" if num <= 4 else "Big"
    if sz != exp_sz:
        rule_violations.append((seq, pid, "size", sz, exp_sz))
        
    if num == 0: exp_col = "Red+Violet"
    elif num == 5: exp_col = "Green+Violet"
    elif num in [1, 3, 7, 9]: exp_col = "Green"
    else: exp_col = "Red"
    
    if col != exp_col:
        rule_violations.append((seq, pid, "colour", col, exp_col))

print(f"Rule violations: {len(rule_violations)}")

# 2. Check ML_Ready
ws_ml = wb["ML_Ready"]
ml_rows = list(ws_ml.iter_rows(min_row=2, values_only=True))
print(f"Total ML_Ready rows: {len(ml_rows)} (Expected: 1610)")

raw_dict = {str(r[1].value): (r[3].value, r[4].value, r[5].value) for r in raw_rows}
raw_list = [(str(r[1].value), r[3].value, r[4].value, r[5].value) for r in raw_rows]
pid_to_idx = {r[0]: idx for idx, r in enumerate(raw_list)}

ml_mismatches = []
gap_crossings = []

for row in ml_rows:
    t_pid = str(row[1])
    t_num = row[18]
    t_col = row[19]
    t_sz = row[20]
    
    if raw_dict[t_pid] != (t_num, t_col, t_sz):
        ml_mismatches.append((t_pid, 'target'))
        
    t_idx = pid_to_idx[t_pid]
    for k in range(1, 6):
        prev_num = row[3 + (k - 1) * 3]
        prev_col = row[4 + (k - 1) * 3]
        prev_sz = row[5 + (k - 1) * 3]
        
        expected_prev_pid, exp_num, exp_col, exp_sz = raw_list[t_idx - k]
        if (prev_num, prev_col, prev_sz) != (exp_num, exp_col, exp_sz):
            ml_mismatches.append((t_pid, f'prev_{k}'))
            
        cur_pid_step = raw_list[t_idx - k + 1][0]
        prev_pid_step = expected_prev_pid
        if not is_consecutive_period(prev_pid_step, cur_pid_step):
            gap_crossings.append((cur_pid_step, prev_pid_step))

print(f"ML value mismatches: {len(ml_mismatches)}")
print(f"ML gap crossings: {len(gap_crossings)}")

# 3. Check Summary
ws_sum = wb["Summary"]
raw_col_counts = Counter(r[4].value for r in raw_rows)
print("Raw_Data colour counts:", dict(raw_col_counts))

# Check timestamps
earliest = raw_rows[0]
latest = raw_rows[-1]
print(f"Earliest: Seq {earliest[0].value}, PID {earliest[1].value}, Time: {earliest[2].value}")
print(f"Latest: Seq {latest[0].value}, PID {latest[1].value}, Time: {latest[2].value}")
