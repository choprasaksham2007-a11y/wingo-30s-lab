import os
import glob
import cv2
import openpyxl
from collections import defaultdict
import numpy as np
import asyncio
from PIL import Image
import winocr

zip_dir = r"extraction\extracted_zip"
files = sorted(glob.glob(os.path.join(zip_dir, "*.jpeg")))

target_seqs = [528, 598, 647, 677, 695, 696, 716, 773, 876, 885, 895, 905, 915, 955, 965, 975, 985, 1006, 1026, 1036, 1095, 1105, 1123, 1163, 1222, 1232]

# Load existing Raw_Data to get Period IDs for these 26 sequence numbers
excel_path = r"project_reports\Colour_Prediction_Dataset.xlsx"
wb = openpyxl.load_workbook(excel_path, data_only=True)
ws_raw = wb["Raw_Data"]

headers = [c.value for c in ws_raw[1]]
seq_idx = headers.index("sequence_no")
pid_idx = headers.index("Period ID")
num_idx = headers.index("number")
col_idx = headers.index("colour")
size_idx = headers.index("size")
ts_idx = headers.index("timestamp")

target_pids = {}
for row in ws_raw.iter_rows(min_row=2, values_only=True):
    seq = row[seq_idx]
    if seq in target_seqs:
        target_pids[row[pid_idx]] = {
            "seq": seq,
            "period_id": row[pid_idx],
            "timestamp": row[ts_idx],
            "number": row[num_idx],
            "colour": row[col_idx],
            "size": row[size_idx]
        }

print(f"Target Period IDs count: {len(target_pids)}")

# Find which screenshots contain each target Period ID
async def find_sources():
    pid_to_files = defaultdict(list)
    for f in files:
        fname = os.path.basename(f)
        im = cv2.imread(f)
        pid_strip = im[580:1350, 45:280]
        res = await winocr.recognize_pil(Image.fromarray(cv2.cvtColor(pid_strip, cv2.COLOR_BGR2RGB)))
        for line in res.lines:
            txt = "".join([w.text.strip() for w in line.words]).replace(" ", "").replace("o", "0").replace("O", "0")
            for t_pid in target_pids:
                if t_pid in txt:
                    w0 = line.words[0]
                    y_rel = w0.bounding_rect.y + w0.bounding_rect.height / 2.0
                    y_full = 580 + y_rel
                    pid_to_files[t_pid].append({
                        "file": fname,
                        "y_full": y_full,
                        "line_txt": txt
                    })
    return pid_to_files

pid_to_files = asyncio.run(find_sources())

for pid, info in sorted(target_pids.items(), key=lambda x: x[1]["seq"]):
    srcs = pid_to_files.get(pid, [])
    src_names = [s["file"] for s in srcs]
    print(f"Seq {info['seq']:4d} | PID {pid} | Num {info['number']} | Col: {info['colour']} | Sources ({len(srcs)}): {src_names}")

# Save this mapping
import json
with open(r"extraction\target_26_sources.json", "w") as jf:
    json.dump({pid: {"info": target_pids[pid], "sources": pid_to_files.get(pid, [])} for pid in target_pids}, jf, indent=2)

print("\nSaved target_26_sources.json")
