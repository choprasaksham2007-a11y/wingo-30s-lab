import sys
import openpyxl
from collections import Counter

sys.path.append(r"extraction")
from src.ml_dataset_builder import is_consecutive_period

orig_path = r"project_reports\Colour_Prediction_Dataset.xlsx"
corr_path = r"project_reports\Colour_Prediction_Dataset_corrected.xlsx"

wb_orig = openpyxl.load_workbook(orig_path, data_only=True)
wb_corr = openpyxl.load_workbook(corr_path, data_only=True)

ws_orig_raw = wb_orig["Raw_Data"]
ws_corr_raw = wb_corr["Raw_Data"]

print("=" * 65)
print("RUNNING 5 COMPREHENSIVE VERIFICATION CHECKS")
print("=" * 65)

# -------------------------------------------------------------
# Check 1: 1,260 rows, all Period IDs unique, 17 digits, stored as text.
# -------------------------------------------------------------
raw_rows = list(ws_corr_raw.iter_rows(min_row=2, values_only=False))
pids = [cell[1].value for cell in raw_rows]
pid_types = [cell[1].data_type for cell in raw_rows]

c1_count = len(raw_rows) == 1260
c1_unique = len(set(pids)) == 1260
c1_digits = all(len(str(p)) == 17 and str(p).isdigit() for p in pids)
c1_text = all(t == 's' for t in pid_types)

c1_pass = c1_count and c1_unique and c1_digits and c1_text
print(f"Check 1: Total rows = {len(raw_rows)} (Expected 1260): {'PASS' if c1_count else 'FAIL'}")
print(f"Check 1: Unique Period IDs = {len(set(pids))}: {'PASS' if c1_unique else 'FAIL'}")
print(f"Check 1: All Period IDs 17 digits: {'PASS' if c1_digits else 'FAIL'}")
print(f"Check 1: All Period IDs stored as text: {'PASS' if c1_text else 'FAIL'}")
print(f"==> CHECK 1 RESULT: {'PASS' if c1_pass else 'FAIL'}\n")

# -------------------------------------------------------------
# Check 2: Every colour and size follows the number rules (0 violations).
# -------------------------------------------------------------
rule_violations = []
for r in raw_rows:
    seq = r[0].value
    pid = r[1].value
    num = r[3].value
    col = r[4].value
    sz = r[5].value
    
    exp_sz = 'Small' if num <= 4 else 'Big'
    if sz != exp_sz:
        rule_violations.append((seq, 'size', sz, exp_sz))
        
    if num == 0:
        exp_col = 'Red+Violet'
    elif num == 5:
        exp_col = 'Green+Violet'
    elif num in [1, 3, 7, 9]:
        exp_col = 'Green'
    else:
        exp_col = 'Red'
    if col != exp_col:
        rule_violations.append((seq, 'colour', col, exp_col))

c2_pass = len(rule_violations) == 0
print(f"Check 2: Rule violations = {len(rule_violations)}")
print(f"==> CHECK 2 RESULT: {'PASS' if c2_pass else 'FAIL'}\n")

# -------------------------------------------------------------
# Check 3: The colour counts in Summary match the real counts in Raw_Data.
# -------------------------------------------------------------
col_counts_raw = Counter([r[4].value for r in raw_rows])
ws_sum = wb_corr["Summary"]
sum_col_counts = {}
for r in range(1, 35):
    lbl = ws_sum.cell(r, 5).value
    if lbl in ['Green', 'Red', 'Green+Violet', 'Red+Violet']:
        sum_col_counts[lbl] = ws_sum.cell(r, 6).value

c3_match = (dict(col_counts_raw) == sum_col_counts)
expected_counts = {'Green': 501, 'Red': 500, 'Green+Violet': 135, 'Red+Violet': 124}
c3_exact = (sum_col_counts == expected_counts)
c3_pass = c3_match and c3_exact
print(f"Check 3: Raw_Data colour counts = {dict(col_counts_raw)}")
print(f"Check 3: Summary sheet counts = {sum_col_counts}")
print(f"Check 3: Summary matches Raw_Data and target (501, 500, 135, 124): {'PASS' if c3_pass else 'FAIL'}")
print(f"==> CHECK 3 RESULT: {'PASS' if c3_pass else 'FAIL'}\n")

# -------------------------------------------------------------
# Check 4: ML_Ready has exactly 1230 rows, every target and every prev_k value matches Raw_Data, and no row uses rounds across a gap.
# -------------------------------------------------------------
ws_ml = wb_corr["ML_Ready"]
ml_rows = list(ws_ml.iter_rows(min_row=2, values_only=True))
c4_count = len(ml_rows) == 1230

raw_dict = {str(r[1].value): (r[3].value, r[4].value, r[5].value) for r in raw_rows}
raw_list = [(str(r[1].value), r[3].value, r[4].value, r[5].value) for r in raw_rows]
pid_to_idx = {r[0]: idx for idx, r in enumerate(raw_list)}

ml_mismatches = []
gap_crossings = []

for row in ml_rows:
    t_pid = str(row[1])
    t_num = row[18]
    t_col = row[19]
    t_sz = row[20]
    
    # Check target match with Raw_Data
    if raw_dict[t_pid] != (t_num, t_col, t_sz):
        ml_mismatches.append((t_pid, 'target'))
        
    t_idx = pid_to_idx[t_pid]
    for k in range(1, 6):
        prev_num = row[3 + (k - 1) * 3]
        prev_col = row[4 + (k - 1) * 3]
        prev_sz = row[5 + (k - 1) * 3]
        
        expected_prev_pid, exp_num, exp_col, exp_sz = raw_list[t_idx - k]
        if (prev_num, prev_col, prev_sz) != (exp_num, exp_col, exp_sz):
            ml_mismatches.append((t_pid, f'prev_{k}'))
            
        cur_pid_step = raw_list[t_idx - k + 1][0]
        prev_pid_step = expected_prev_pid
        if not is_consecutive_period(prev_pid_step, cur_pid_step):
            gap_crossings.append((cur_pid_step, prev_pid_step))

c4_pass = c4_count and (len(ml_mismatches) == 0) and (len(gap_crossings) == 0)
print(f"Check 4: ML_Ready row count = {len(ml_rows)} (Expected 1230): {'PASS' if c4_count else 'FAIL'}")
print(f"Check 4: ML lag & target value mismatches = {len(ml_mismatches)}: {'PASS' if len(ml_mismatches)==0 else 'FAIL'}")
print(f"Check 4: ML gap crossings = {len(gap_crossings)}: {'PASS' if len(gap_crossings)==0 else 'FAIL'}")
print(f"==> CHECK 4 RESULT: {'PASS' if c4_pass else 'FAIL'}\n")

# -------------------------------------------------------------
# Check 5: Exactly 26 colour values were changed, and every other cell is identical to the original.
# -------------------------------------------------------------
orig_rows = list(ws_orig_raw.iter_rows(min_row=2, values_only=True))
changed_cols = []
other_diffs = []
for i in range(len(raw_rows)):
    o = orig_rows[i]
    c = [cell.value for cell in raw_rows[i]]
    
    # Metadata check: sequence_no, Period ID, timestamp, number, size
    if (o[0] != c[0]) or (str(o[1]) != str(c[1])) or (str(o[2]) != str(c[2])) or (o[3] != c[3]) or (o[5] != c[5]):
        other_diffs.append((o[0], 'metadata mismatch'))
        
    if o[4] != c[4]:
        changed_cols.append((o[0], o[1], o[4], c[4]))

target_seqs = [528, 598, 647, 677, 695, 696, 716, 773, 876, 885, 895, 905, 915, 955, 965, 975, 985, 1006, 1026, 1036, 1095, 1105, 1123, 1163, 1222, 1232]
changed_seqs = [x[0] for x in changed_cols]

c5_exact_26 = (len(changed_cols) == 26)
c5_exact_matches = (sorted(changed_seqs) == sorted(target_seqs))
c5_no_other_diffs = (len(other_diffs) == 0)
c5_pass = c5_exact_26 and c5_exact_matches and c5_no_other_diffs

print(f"Check 5: Exactly 26 colour values changed = {len(changed_cols)}: {'PASS' if c5_exact_26 else 'FAIL'}")
print(f"Check 5: Changed sequences match user target list: {'PASS' if c5_exact_matches else 'FAIL'}")
print(f"Check 5: Other cell differences = {len(other_diffs)}: {'PASS' if c5_no_other_diffs else 'FAIL'}")
print(f"==> CHECK 5 RESULT: {'PASS' if c5_pass else 'FAIL'}\n")

print("=" * 65)
all_pass = c1_pass and c2_pass and c3_pass and c4_pass and c5_pass
print(f"OVERALL VALIDATION: {'ALL 5 CHECKS PASSED PERFECTLY!' if all_pass else 'SOME CHECKS FAILED'}")
print("=" * 65)
