import pickle
from collections import defaultdict

with open(r"extraction\data\aligned_v2_observations.pkl", "rb") as pf:
    all_obs = pickle.load(pf)

grouped = defaultdict(list)
for obs in all_obs:
    grouped[obs.period_id].append(obs)

unique_pids = sorted(list(grouped.keys()), key=lambda x: (len(x), x))
print(f"Total unique Period IDs in v2: {len(unique_pids)}")

conflicts = 0
for pid, obs_list in grouped.items():
    nums = set(o.number for o in obs_list if o.number is not None)
    if len(nums) > 1:
        conflicts += 1
        print(f"Conflict in PID {pid}: numbers observed = {nums}")
        for o in obs_list:
            print(f"   file: {o.source_screenshot}, num={o.number}, size={o.big_small}, conf={o.ocr_confidence}")

print(f"\nTOTAL NUMBER CONFLICTS IN V2: {conflicts}")
