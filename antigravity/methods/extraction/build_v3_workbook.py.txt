def get_timestamp(pid: str) -> str:
    # 52278 = 2026-10-24 00:29:00
    anchor = datetime(2026, 10, 24, 0, 29, 0)
    date_part = pid[:8]
    suffix = int(pid[-5:])
    if date_part == "20261023":
        diff = suffix - 52278
        dt = anchor + timedelta(seconds=diff * 30)
    elif date_part == "20261022":
        # 50001 on 20261023 is (50001 - 52278) = -2277 rounds before anchor
        dt_50001 = anchor + timedelta(seconds=(50001 - 52278) * 30)
        dt_52880 = dt_50001 - timedelta(seconds=30)
        diff_22 = 52880 - suffix
        dt = dt_52880 - timedelta(seconds=diff_22 * 30)
    else:
        return ""
    return dt.strftime("%Y-%m-%d %H:%M:%S")

def derive_colour(n):
    if n == 0: return "Red+Violet"
    if n == 5: return "Green+Violet"
    if n in [1,3,7,9]: return "Green"
    return "Red"

def derive_size(n):
    return "Small" if n <= 4 else "Big"

# ── Load existing v2 records (1,640 rounds) ──
v2_path = r"project_reports\Colour_Prediction_Dataset_v2.xlsx"
wb_v2 = openpyxl.load_workbook(v2_path, data_only=True)
ws_v2_raw = wb_v2["Raw_Data"]

existing_records = []
for row in ws_v2_raw.iter_rows(min_row=2, values_only=True):
    existing_records.append({
        "seq": int(row[0]),
        "pid": str(row[1]),
        "ts": str(row[2]),
        "num": int(row[3]),
        "col": str(row[4]),
        "size": str(row[5]),
        "col_orig": str(row[6]),
        "fix_note": str(row[7])
    })

print(f"Loaded {len(existing_records)} existing records from v2.")

# ── Load Batch 3 records (495 rounds) ──
with open(r"extraction\reconciled_batch3.json") as f:
    b3_data = json.load(f)

existing_pids = set(r["pid"] for r in existing_records)
new_pids = sorted([p for p in b3_data if p not in existing_pids])
print(f"New unique rounds from Batch 3: {len(new_pids)}")

new_records = []
for pid in new_pids:
    b3 = b3_data[pid]
    new_records.append({
        "seq": 0,
        "pid": pid,
        "ts": get_timestamp(pid),
        "num": b3["number"],
        "col": derive_colour(b3["number"]),
        "size": derive_size(b3["number"]),
        "col_orig": derive_colour(b3["number"]),
        "fix_note": "Extracted from batch 3 screenshots (night session)"
    })

# ── Merge and sort chronologically ──
all_records = existing_records + new_records
all_records.sort(key=lambda r: (r["pid"][:8], int(r["pid"][-5:])))
for i, rec in enumerate(all_records):
    rec["seq"] = i + 1

total_rows = len(all_records)
print(f"Total merged records in v3: {total_rows}")

# ── Identify gaps for Missing_Rounds ──
gaps = []
for i in range(len(all_records) - 1):
    p1 = all_records[i]["pid"]
    p2 = all_records[i+1]["pid"]
    if not is_consecutive_period(p1, p2):
        d1, d2 = p1[:8], p2[:8]
        s1, s2 = int(p1[-5:]), int(p2[-5:])
        if d1 == d2:
            missing = s2 - s1 - 1
        elif d1 == "20261022" and d2 == "20261023":
            missing = (52880 - s1) + (s2 - 50001)
        else:
            missing = 0
        gaps.append({
            "gap_num": len(gaps) + 1,
            "before_pid": p1,
            "after_pid": p2,
            "before_seq": all_records[i]["seq"],
            "after_seq": all_records[i+1]["seq"],
            "missing_count": missing,
            "before_ts": all_records[i]["ts"],
            "after_ts": all_records[i+1]["ts"]
        })

print(f"Total gaps in v3: {len(gaps)}")
for g in gaps:
    print(f"  Gap {g['gap_num']}: {g['before_pid']} ({g['before_ts']}) -> {g['after_pid']} ({g['after_ts']}): {g['missing_count']} missing rounds")

# ── Build ML_Ready ──
ml_rows = []
ml_seq = 1
for i in range(5, len(all_records)):
    target = all_records[i]
    window = [all_records[i-k] for k in range(5, -1, -1)]
    has_gap = False
    for w in range(len(window)-1):
        if not is_consecutive_period(window[w]["pid"], window[w+1]["pid"]):
            has_gap = True
            break
    if has_gap:
        continue
    lags = [all_records[i-k] for k in range(1, 6)]
    ml_rows.append({
        "ml_seq": ml_seq,
        "target": target,
        "lags": lags
    })
    ml_seq += 1

print(f"ML_Ready rows in v3: {len(ml_rows)}")
