import pickle
import openpyxl

# Load existing 500
excel_path = r"project_reports\inputs\input_9f7c60764d.xlsx"
wb = openpyxl.load_workbook(excel_path)
ws_raw = wb["Raw_Data"]

existing_pids = set()
for row in ws_raw.iter_rows(min_row=2, values_only=True):
    pid = row[1]
    if pid is not None:
        existing_pids.add(str(pid).strip().replace('\u200b', '').replace('\ufeff', ''))

# Load new v2
with open(r"extraction\data\aligned_v2_observations.pkl", "rb") as pf:
    new_obs = pickle.load(pf)

new_pids = set(o.period_id for o in new_obs)

combined_pids = existing_pids.union(new_pids)
print(f"Existing unique PIDs: {len(existing_pids)}")
print(f"New unique PIDs: {len(new_pids)}")
print(f"Total Combined unique PIDs: {len(combined_pids)}")

# Check continuity in 20261022
d22_pids = sorted([p for p in combined_pids if p.startswith("20261022")], key=lambda x: int(x))
print(f"20261022 range: {d22_pids[0]} to {d22_pids[-1]}, total={len(d22_pids)}")
min_val = int(d22_pids[0])
max_val = int(d22_pids[-1])
expected_count = max_val - min_val + 1
print(f"20261022 expected consecutive count: {expected_count}")
missing_22 = []
present_set_22 = set(int(p) for p in d22_pids)
for v in range(min_val, max_val + 1):
    if v not in present_set_22:
        missing_22.append(str(v))
print(f"Missing in 20261022: {len(missing_22)}")
if missing_22:
    print(f"  First 5 missing: {missing_22[:5]}")
    print(f"  Last 5 missing: {missing_22[-5:]}")

# Check continuity in 20261023
d23_pids = sorted([p for p in combined_pids if p.startswith("20261023")], key=lambda x: int(x))
print(f"\n20261023 range: {d23_pids[0]} to {d23_pids[-1]}, total={len(d23_pids)}")
min_val_23 = int(d23_pids[0])
max_val_23 = int(d23_pids[-1])
expected_count_23 = max_val_23 - min_val_23 + 1
print(f"20261023 expected consecutive count: {expected_count_23}")
missing_23 = []
present_set_23 = set(int(p) for p in d23_pids)
for v in range(min_val_23, max_val_23 + 1):
    if v not in present_set_23:
        missing_23.append(str(v))
print(f"Missing in 20261023: {len(missing_23)}")
if missing_23:
    print(f"  First 5 missing: {missing_23[:5]}")
    print(f"  Last 5 missing: {missing_23[-5:]}")
