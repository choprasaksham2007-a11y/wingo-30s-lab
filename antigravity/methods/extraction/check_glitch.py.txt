import cv2
import asyncio
from PIL import Image
import winocr

img = cv2.imread(r"extraction\extracted_zip\WhatsApp Image 2026-10-23 at 09.34.09.jpeg")

async def main():
    table_crop = img[500:1400, 40:680]
    # row 5 is at index 5 (6th row): y around 500 + 75 + 5*76.6 = 958
    row_crop = img[940:1020, 40:300]
    pil_im = Image.fromarray(cv2.cvtColor(row_crop, cv2.COLOR_BGR2RGB))
    res = await winocr.recognize_pil(pil_im)
    print("Row 5 PID crop OCR:", res.text.strip())

asyncio.run(main())
