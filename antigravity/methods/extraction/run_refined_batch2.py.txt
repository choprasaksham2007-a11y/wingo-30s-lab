import os
import glob
import cv2
import asyncio
from PIL import Image
import winocr
import numpy as np
import json
from collections import defaultdict, Counter
from datetime import datetime, timedelta

batch2_dir = r"extraction\extracted_zip_batch2"
templates = np.load(r"extraction\digit_templates.npy", allow_pickle=True).item()

def get_timestamp(pid: str) -> str:
    # Anchor: 20261023100050494 = 2026-10-23 09:37:00 AM
    anchor = datetime(2026, 10, 23, 9, 37, 0)
    suffix = int(pid[-5:])
    diff = suffix - 50494
    dt = anchor + timedelta(seconds=diff * 30)
    return dt.strftime("%Y-%m-%d %H:%M:%S")

def is_top_right_closed(num_crop):
    hsv = cv2.cvtColor(num_crop, cv2.COLOR_BGR2HSV)
    # Red or Green mask
    r_mask = cv2.inRange(hsv, np.array([0, 70, 70]), np.array([10, 255, 255])) | cv2.inRange(hsv, np.array([170, 70, 70]), np.array([180, 255, 255]))
    # top right quadrant: y: 5 to 15, x: 22 to 32
    return cv2.countNonZero(r_mask[5:15, 22:32]) > 30

def classify_digit_refined(num_crop, bs_val):
    scores = {}
    for digit, tpl in templates.items():
        res = cv2.matchTemplate(num_crop, tpl, cv2.TM_CCOEFF_NORMED)
        scores[int(digit)] = float(cv2.minMaxLoc(res)[1])
        
    # HSV analysis
    hsv = cv2.cvtColor(num_crop, cv2.COLOR_BGR2HSV)
    g_cnt = cv2.countNonZero(cv2.inRange(hsv, np.array([35, 50, 50]), np.array([85, 255, 255])))
    r_cnt = cv2.countNonZero(cv2.inRange(hsv, np.array([0, 70, 70]), np.array([10, 255, 255])) | cv2.inRange(hsv, np.array([170, 70, 70]), np.array([180, 255, 255])))
    v_cnt = cv2.countNonZero(cv2.inRange(hsv, np.array([125, 40, 40]), np.array([165, 255, 255])))
    
    # 0 vs 5:
    if v_cnt > 80:
        if r_cnt > 80:
            return 0, "Red+Violet", "Small", scores[0]
        else:
            return 5, "Green+Violet", "Big", scores[5]
            
    is_green = (g_cnt > 150 and r_cnt < 80)
    is_red = (r_cnt > 150 and g_cnt < 80)
    
    if is_green:
        # Possible: 1, 3, 7, 9
        # Check size if available
        if bs_val == "Small":
            # 1 vs 3
            best_d = 1 if scores[1] > scores[3] else 3
            return best_d, "Green", "Small", scores[best_d]
        elif bs_val == "Big":
            # 7 vs 9
            best_d = 7 if scores[7] > scores[9] else 9
            return best_d, "Green", "Big", scores[best_d]
        else:
            # Fallback to score
            cand = [1, 3, 7, 9]
            best_d = max(cand, key=lambda d: scores[d])
            sz = "Small" if best_d in [1, 3] else "Big"
            return best_d, "Green", sz, scores[best_d]
            
    if is_red:
        # Possible: 2, 4, 6, 8
        if bs_val == "Small":
            best_d = 2 if scores[2] > scores[4] else 4
            return best_d, "Red", "Small", scores[best_d]
        elif bs_val == "Big":
            # 6 vs 8: check top-right closure
            if is_top_right_closed(num_crop):
                return 8, "Red", "Big", scores[8]
            else:
                return 6, "Red", "Big", scores[6]
        else:
            cand = [2, 4, 6, 8]
            best_d = max(cand, key=lambda d: scores[d])
            if best_d in [6, 8]:
                best_d = 8 if is_top_right_closed(num_crop) else 6
            sz = "Small" if best_d in [2, 4] else "Big"
            return best_d, "Red", sz, scores[best_d]
            
    # Generic fallback
    best_d = max(scores.keys(), key=lambda d: scores[d])
    sz = "Small" if best_d <= 4 else "Big"
    if best_d == 0: col = "Red+Violet"
    elif best_d == 5: col = "Green+Violet"
    elif best_d in [1, 3, 7, 9]: col = "Green"
    else: col = "Red"
    return best_d, col, sz, scores[best_d]

async def process_batch2():
    files = sorted(glob.glob(os.path.join(batch2_dir, "*.jpeg")))
    obs_by_pid = defaultdict(list)
    
    for f in files:
        fname = os.path.basename(f)
        if fname in ['WhatsApp Image 2026-10-23 at 22.06.44.jpeg', 'WhatsApp Image 2026-10-23 at 22.06.45.jpeg']:
            continue
            
        img = cv2.imread(f)
        pid_strip = img[550:1400, 30:290]
        res_pid = await winocr.recognize_pil(Image.fromarray(cv2.cvtColor(pid_strip, cv2.COLOR_BGR2RGB)))
        
        bs_strip = img[550:1400, 440:540]
        res_bs = await winocr.recognize_pil(Image.fromarray(cv2.cvtColor(bs_strip, cv2.COLOR_BGR2RGB)))
        
        bs_lines = []
        for l in res_bs.lines:
            txt = " ".join([w.text.strip() for w in l.words])
            if "Big" in txt or "Small" in txt or "sma" in txt.lower():
                val = "Small" if ("Small" in txt or "sma" in txt.lower()) else "Big"
                y_c = l.words[0].bounding_rect.y + l.words[0].bounding_rect.height / 2.0
                bs_lines.append((y_c, val))
                
        for line in res_pid.lines:
            txt = "".join([w.text.strip() for w in line.words]).replace(" ", "").replace("o", "0").replace("O", "0")
            for i in range(len(txt) - 16):
                chunk = txt[i:i+17]
                if chunk.isdigit() and chunk.startswith("202610231000"):
                    w0 = line.words[0]
                    y_rel = w0.bounding_rect.y + w0.bounding_rect.height / 2.0
                    yc = int(round(550 + y_rel))
                    
                    best_bs = None
                    min_dist = 999.0
                    for bs_y, bs_val in bs_lines:
                        dist = abs(bs_y - y_rel)
                        if dist < min_dist and dist < 35:
                            min_dist = dist
                            best_bs = bs_val
                            
                    num_crop = img[yc-20:yc+20, 345:385]
                    d, col, sz, conf = classify_digit_refined(num_crop, best_bs)
                    
                    obs_by_pid[chunk].append({
                        "file": fname,
                        "yc": yc,
                        "digit": d,
                        "col": col,
                        "size": sz,
                        "bs_ocr": best_bs,
                        "conf": round(conf, 3)
                    })
                    break
                    
    return obs_by_pid

obs_by_pid = asyncio.run(process_batch2())

reconciled = {}
for pid in sorted(obs_by_pid.keys()):
    obs = obs_by_pid[pid]
    # Check consensus on digit
    digits = [o["digit"] for o in obs]
    best_d = Counter(digits).most_common(1)[0][0]
    
    # Derivations
    sz = "Small" if best_d <= 4 else "Big"
    if best_d == 0: col = "Red+Violet"
    elif best_d == 5: col = "Green+Violet"
    elif best_d in [1, 3, 7, 9]: col = "Green"
    else: col = "Red"
    
    ts = get_timestamp(pid)
    
    reconciled[pid] = {
        "period_id": pid,
        "timestamp": ts,
        "number": best_d,
        "colour": col,
        "size": sz,
        "source_count": len(obs),
        "sources": [o["file"] for o in obs]
    }

print(f"Total extracted and verified rounds in batch 2: {len(reconciled)}")
with open(r"extraction\reconciled_batch2_refined.json", "w") as f:
    json.dump(reconciled, f, indent=2)
print("Saved reconciled_batch2_refined.json successfully.")
